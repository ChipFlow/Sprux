17// allows to retrieve a block in a coalesced-block matrix
18struct CoalescedAccessor {
19 // no constructor, to be able to use it as argument of a Cuda kernel
20 void init(const int64_t* spanStart_, const int64_t* spanToLump_, const int64_t* lumpStart_,
21 const int64_t* spanOffsetInLump_, const int64_t* chainColPtr_,
22 const int64_t* chainRowSpan_, const int64_t* chainData_) {
23 spanStart = spanStart_;
24 spanToLump = spanToLump_;
25 lumpStart = lumpStart_;
26 spanOffsetInLump = spanOffsetInLump_;
27 chainColPtr = chainColPtr_;
28 chainRowSpan = chainRowSpan_;
29 chainData = chainData_;
30 }
31
32 // returns size of i-th parameter
33 __BASPACHO_HOST_DEVICE__
34 int64_t paramSize(int64_t blockIndex) const {
35 return spanStart[blockIndex + 1] - spanStart[blockIndex];
36 }
37
38 // returns start of i-th parameter
39 __BASPACHO_HOST_DEVICE__
40 int64_t paramStart(int64_t blockIndex) const { return spanStart[blockIndex]; }
41
42 // return: pair (offset, stride) to identify block in numeric factor data
43 __BASPACHO_HOST_DEVICE__
44 std::pair<int64_t, int64_t> blockOffset(int64_t rowBlockIndex, int64_t colBlockIndex) const {
45 BASPACHO_CHECK_GE(rowBlockIndex, colBlockIndex);
46 int64_t lump = spanToLump[colBlockIndex];
47 int64_t lumpSize = lumpStart[lump + 1] - lumpStart[lump];
48 int64_t offsetInLump = spanOffsetInLump[colBlockIndex];
49 int64_t start = chainColPtr[lump];
50 int64_t end = chainColPtr[lump + 1];
51 // bisect to find `rowBlockIndex` in chainRowSpan[start:end]
52 int64_t pos = bisect(chainRowSpan + start, end - start, rowBlockIndex);
53 BASPACHO_CHECK_EQ(chainRowSpan[start + pos], rowBlockIndex);
54 return std::make_pair(chainData[start + pos] + offsetInLump, lumpSize);
55 }
56
57 // return: pair (offset, stride) to identify diagonal block in numeric factor data
58 __BASPACHO_HOST_DEVICE__
59 std::pair<int64_t, int64_t> diagBlockOffset(int64_t blockIndex) const {
60 int64_t lump = spanToLump[blockIndex];
61 int64_t lumpSize = lumpStart[lump + 1] - lumpStart[lump];
62 int64_t offsetInLump = spanOffsetInLump[blockIndex];
63 int64_t start = chainColPtr[lump];
64 return std::make_pair(chainData[start] + offsetInLump * (1 + lumpSize), lumpSize);
65 }
66
67 // return: block reference, from factor data pointer
68 template <int rowSize = Eigen::Dynamic, int64_t colSize = Eigen::Dynamic, typename T>
69 __BASPACHO_HOST_DEVICE__ auto block(T* data, int64_t rowBlockIndex, int64_t colBlockIndex) const {
70 using namespace Eigen;
71 auto offsetStride = blockOffset(rowBlockIndex, colBlockIndex);
72 auto offset = std::get<0>(offsetStride);
73 auto stride = std::get<1>(offsetStride);
74 if (rowSize != Dynamic) {
75 BASPACHO_CHECK_EQ(rowSize, paramSize(rowBlockIndex));
76 }
77 if (colSize != Dynamic) {
78 BASPACHO_CHECK_EQ(colSize, paramSize(colBlockIndex));
79 }
80 return Map<Matrix<T, rowSize, colSize, RowMajor>, 0, OuterStride<>>(
81 data + offset, rowSize != Dynamic ? rowSize : paramSize(rowBlockIndex),
82 colSize != Dynamic ? colSize : paramSize(colBlockIndex), OuterStride<>(stride));
83 }
84
85 // return: diagonal block reference, from factor data pointer
86 template <int size = Eigen::Dynamic, typename T>
87 __BASPACHO_HOST_DEVICE__ auto diagBlock(T* data, int64_t blockIndex) const {
88 using namespace Eigen;
89 auto offsetStride = diagBlockOffset(blockIndex);
90 auto offset = std::get<0>(offsetStride);
91 auto stride = std::get<1>(offsetStride);
92 if (size != Dynamic) {
93 BASPACHO_CHECK_EQ(size, paramSize(blockIndex));
94 }
95 int pSize = size != Dynamic ? size : paramSize(blockIndex);
96 return Map<Matrix<T, size, size, RowMajor>, 0, OuterStride<>>(data + offset, pSize, pSize,
97 OuterStride<>(stride));
98 }
99
100 const int64_t* spanStart;
101 const int64_t* spanToLump;
102 const int64_t* lumpStart;
103 const int64_t* spanOffsetInLump;
104 const int64_t* chainColPtr;
105 const int64_t* chainRowSpan;
106 const int64_t* chainData;
107};

143 // return: pair (offset, stride) to identify block in numeric factor data
144 __BASPACHO_HOST_DEVICE__
145 std::tuple<int64_t, int64_t, bool> blockOffset(int64_t rowBlockIndex,
146 int64_t colBlockIndex) const {
147 int64_t permRowBlockIndex = permutation[rowBlockIndex];
148 int64_t permColBlockIndex = permutation[colBlockIndex];
149 auto offsetStride = plainAcc.blockOffset(std::max(permRowBlockIndex, permColBlockIndex),
150 std::min(permRowBlockIndex, permColBlockIndex));
151 auto offset = std::get<0>(offsetStride);
152 auto stride = std::get<1>(offsetStride);
153 bool flipped = permRowBlockIndex < permColBlockIndex;
154 return std::make_tuple(offset, stride, flipped);
155 }
156
157 // return: pair (offset, stride) to identify diagonal block in numeric factor data
158 __BASPACHO_HOST_DEVICE__
159 std::pair<int64_t, int64_t> diagBlockOffset(int64_t blockIndex) const {
160 return plainAcc.diagBlockOffset(permutation[blockIndex]);
161 }

163 // return: block reference, from factor data pointer
164 template <int rowSize = Eigen::Dynamic, int64_t colSize = Eigen::Dynamic, typename T>
165 __BASPACHO_HOST_DEVICE__ auto block(T* data, int64_t rowBlockIndex, int64_t colBlockIndex) const {
166 using namespace Eigen;
167 if (rowSize != Dynamic) {
168 BASPACHO_CHECK_EQ(rowSize, paramSize(rowBlockIndex));
169 }
170 if (colSize != Dynamic) {
171 BASPACHO_CHECK_EQ(colSize, paramSize(colBlockIndex));
172 }
173 auto offsetStrideFlip = blockOffset(rowBlockIndex, colBlockIndex);
174 auto offset = std::get<0>(offsetStrideFlip);
175 auto stride = std::get<1>(offsetStrideFlip);
176 auto flip = std::get<2>(offsetStrideFlip);
177 return Map<Matrix<T, rowSize, colSize, RowMajor>, 0, Stride<Dynamic, Dynamic>>(
178 data + offset, rowSize != Dynamic ? rowSize : paramSize(rowBlockIndex),
179 colSize != Dynamic ? colSize : paramSize(colBlockIndex),
180 Stride<Dynamic, Dynamic>(flip ? 1 : stride, flip ? stride : 1));
181 }
182
183 // return: diagonal block reference, from factor data pointer
184 template <int size = Eigen::Dynamic, typename T>
185 __BASPACHO_HOST_DEVICE__ auto diagBlock(T* data, int64_t blockIndex) const {
186 using namespace Eigen;

188 auto offset = std::get<0>(offsetStride);
189 auto stride = std::get<1>(offsetStride);
190 if (size != Dynamic) {
191 BASPACHO_CHECK_EQ(size, paramSize(blockIndex));
192 }
193 int pSize = size != Dynamic ? size : paramSize(blockIndex);
194 return Map<Matrix<T, size, size, RowMajor>, 0, OuterStride<>>(data + offset, pSize, pSize,
195 OuterStride<>(stride));
196 }

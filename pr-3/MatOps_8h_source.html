195template <typename T>
196NumericCtxPtr<T> SymbolicCtx::createNumericCtx(int64_t tempBufSize, const T* data) {
197 static const std::type_index T_tIdx(typeid(T));
198 int batchSize = Batch<T>::getSize(data);
199 NumericCtxBase* ctx = createNumericCtxForType(T_tIdx, tempBufSize, batchSize);
200 NumericCtx<T>* typedCtx = dynamic_cast<NumericCtx<T>*>(ctx);
201 BASPACHO_CHECK_NOTNULL(typedCtx);
202 return NumericCtxPtr<T>(typedCtx);
203}
204
205template <typename T>
206SolveCtxPtr<T> SymbolicCtx::createSolveCtx(int nRHS, const T* data) {
207 static const std::type_index T_tIdx(typeid(T));
208 int batchSize = Batch<T>::getSize(data);
209 SolveCtxBase* ctx = createSolveCtxForType(T_tIdx, nRHS, batchSize);
210 SolveCtx<T>* typedCtx = dynamic_cast<SolveCtx<T>*>(ctx);
211 BASPACHO_CHECK_NOTNULL(typedCtx);
212 return SolveCtxPtr<T>(typedCtx);
213}

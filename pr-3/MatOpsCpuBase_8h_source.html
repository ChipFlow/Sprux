50 static int64_t computeMaxBufSize(const CpuBaseSymElimCtx& elim,
51 const CoalescedBlockMatrixSkel& skel, int64_t sRel) {
52 int64_t maxBufferSize = 0;
53
54 // iterate over chains present in this row
55 for (int64_t i = elim.rowPtr[sRel], iEnd = elim.rowPtr[sRel + 1]; i < iEnd; i++) {
56 int64_t lump = elim.colLump[i];
57 int64_t chainColOrd = elim.chainColOrd[i];
58 BASPACHO_CHECK_GE(chainColOrd,
59 1); // there must be a diagonal block
60
61 int64_t ptrStart = skel.chainColPtr[lump] + chainColOrd;
62 int64_t ptrEnd = skel.chainColPtr[lump + 1];
63
64 int64_t nRowsAbove = skel.chainRowsTillEnd[ptrStart - 1];
65 int64_t nRowsChain = skel.chainRowsTillEnd[ptrStart] - nRowsAbove;
66 int64_t nRowsOnward = skel.chainRowsTillEnd[ptrEnd - 1];
67
68 maxBufferSize = std::max(maxBufferSize, nRowsOnward * nRowsChain);
69 }
70
71 return maxBufferSize;
72 }
73
74 virtual SymElimCtxPtr prepareElimination(int64_t lumpsBegin, int64_t lumpsEnd) override {
75 CpuBaseSymElimCtx* elim = new CpuBaseSymElimCtx;
76
77 int64_t spanRowBegin = skel.lumpToSpan[lumpsEnd];
78 int64_t numSpanRows = skel.spanStart.size() - 1 - spanRowBegin;
79 elim->rowPtr.assign(numSpanRows + 1, 0);
80 for (int64_t l = lumpsBegin; l < lumpsEnd; l++) {
81 for (int64_t i = skel.chainColPtr[l], iEnd = skel.chainColPtr[l + 1]; i < iEnd; i++) {
82 int64_t s = skel.chainRowSpan[i];
83 if (s < spanRowBegin) {
84 continue;
85 }
86 int64_t sRel = s - spanRowBegin;
87 elim->rowPtr[sRel]++;
88 }
89 }
90 int64_t totNumChains = cumSumVec(elim->rowPtr);
91 elim->colLump.resize(totNumChains);
92 elim->chainColOrd.resize(totNumChains);
93 for (int64_t l = lumpsBegin; l < lumpsEnd; l++) {
94 for (int64_t iBegin = skel.chainColPtr[l], iEnd = skel.chainColPtr[l + 1], i = iBegin;
95 i < iEnd; i++) {
96 int64_t s = skel.chainRowSpan[i];
97 if (s < spanRowBegin) {
98 continue;
99 }
100 int64_t sRel = s - spanRowBegin;
101 elim->colLump[elim->rowPtr[sRel]] = l;
102 elim->chainColOrd[elim->rowPtr[sRel]] = i - iBegin;
103 elim->rowPtr[sRel]++;
104 }
105 }
106 rewindVec(elim->rowPtr);
107 elim->spanRowBegin = spanRowBegin;
108
109 elim->maxBufferSize = 0;
110 for (int64_t r = 0; r < (int64_t)elim->rowPtr.size() - 1; r++) {
111 elim->maxBufferSize = std::max(elim->maxBufferSize, computeMaxBufSize(*elim, skel, r));
112 }
113 return SymElimCtxPtr(elim);
114 }
115
116 const CoalescedBlockMatrixSkel& skel;
117};

119template <typename T>
120struct CpuBaseNumericCtx : NumericCtx<T> {
121 CpuBaseNumericCtx(const CpuBaseSymbolicCtx& sym, int64_t bufSize, int64_t numSpans)
122 : sym(sym), tempBuffer(bufSize), spanToChainOffset(numSpans) {}
123
124 virtual void potrf(int64_t n, T* data, int64_t offA) override {
125 auto timer = sym.potrfStat.instance(sizeof(T), n);
126 sym.potrfBiggestN = std::max(sym.potrfBiggestN, n);
127
128 Eigen::Map<MatRMaj<T>> matA(data + offA, n, n);
129 Eigen::LLT<Eigen::Ref<MatRMaj<T>>> llt(matA);
130 }
131
132 virtual void trsm(int64_t n, int64_t k, T* data, int64_t offA, int64_t offB) override {
133 auto timer = sym.trsmStat.instance(sizeof(T), n, k);
134
135 using MatCMajD = Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic, Eigen::ColMajor>;
136
137 // col-major's upper = (row-major's lower).transpose()
138 Eigen::Map<const MatCMajD> matA(data + offA, n, n);
139 Eigen::Map<MatRMaj<T>> matB(data + offB, k, n);
140 matA.template triangularView<Eigen::Upper>().template solveInPlace<Eigen::OnTheRight>(matB);
141 }
142
143 virtual void saveSyrkGemm(int64_t m, int64_t n, int64_t k, const T* data,
144 int64_t offset) override {
145 auto timer = sym.sygeStat.instance(sizeof(T), m, n, k);
146 BASPACHO_CHECK_LE(m * n, (int64_t)tempBuffer.size());
147
148 const T* AB = data + offset;
149 T* C = tempBuffer.data();
150 Eigen::Map<const MatRMaj<T>> matA(AB, m, k);
151 Eigen::Map<const MatRMaj<T>> matB(AB, n, k);
152 Eigen::Map<MatRMaj<T>> matC(C, n, m);
153 matC.noalias() = matB * matA.transpose();
154
155 sym.gemmCalls++;
156 }

161 static inline void factorLump(const CoalescedBlockMatrixSkel& skel, T* data, int64_t lump) {
162 int64_t lumpStart = skel.lumpStart[lump];
163 int64_t lumpSize = skel.lumpStart[lump + 1] - lumpStart;
164 int64_t colStart = skel.chainColPtr[lump];
165 int64_t dataPtr = skel.chainData[colStart];
166
167 // in-place lower diag cholesky dec on diagonal block
168 Eigen::Map<MatRMaj<T>> diagBlock(data + dataPtr, lumpSize, lumpSize);
169 { Eigen::LLT<Eigen::Ref<MatRMaj<T>>> llt(diagBlock); }
170
171 int64_t gatheredStart = skel.boardColPtr[lump];
172 int64_t gatheredEnd = skel.boardColPtr[lump + 1];
173 int64_t rowDataStart = skel.boardChainColOrd[gatheredStart + 1];
174 int64_t rowDataEnd = skel.boardChainColOrd[gatheredEnd - 1];
175 int64_t belowDiagStart = skel.chainData[colStart + rowDataStart];
176 int64_t numRows = skel.chainRowsTillEnd[colStart + rowDataEnd - 1] -
177 skel.chainRowsTillEnd[colStart + rowDataStart - 1];
178
179 Eigen::Map<MatRMaj<T>> belowDiagBlock(data + belowDiagStart, numRows, lumpSize);
180 diagBlock.template triangularView<Eigen::Lower>()
181 .transpose()
182 .template solveInPlace<Eigen::OnTheRight>(belowDiagBlock);
183 }

185 static inline void factorSpan(const CoalescedBlockMatrixSkel& skel, T* data, int64_t span) {
186 int64_t lump = skel.spanToLump[span];
187 int64_t spanOffsetInLump = skel.spanOffsetInLump[span];
188 int64_t spanIndexInLump = span - skel.lumpToSpan[lump];
189 int64_t spanSize = skel.spanStart[span + 1] - skel.spanStart[span];
190 int64_t lumpStart = skel.lumpStart[lump];
191 int64_t lumpSize = skel.lumpStart[lump + 1] - lumpStart;
192 int64_t colStart = skel.chainColPtr[lump];
193 int64_t diagBlockPtr = skel.chainData[colStart + spanIndexInLump] + spanOffsetInLump;
194
195 // in-place lower diag cholesky dec on diagonal block
196 OuterStridedMatM<T> diagBlock(data + diagBlockPtr, spanSize, spanSize, OuterStride(lumpSize));
197 { Eigen::LLT<Eigen::Ref<MatRMaj<T>>> llt(diagBlock); }
198
199 int64_t gatheredEnd = skel.boardColPtr[lump + 1];
200 int64_t rowDataEnd = skel.boardChainColOrd[gatheredEnd - 1];
201 int64_t belowDiagPtr = skel.chainData[colStart + spanIndexInLump + 1] + spanOffsetInLump;
202 int64_t numRows = skel.chainRowsTillEnd[colStart + rowDataEnd - 1] -
203 skel.chainRowsTillEnd[colStart + spanIndexInLump];
204
205 OuterStridedMatM<T> belowDiagBlock(data + belowDiagPtr, numRows, spanSize,
206 OuterStride(lumpSize));
207 diagBlock.template triangularView<Eigen::Lower>()
208 .transpose()
209 .template solveInPlace<Eigen::OnTheRight>(belowDiagBlock);
210 }

212 static inline void stridedMatSub(T* dst, int64_t dstStride, const T* src, int64_t srcStride,
213 int64_t rSize, int64_t cSize) {
214 for (uint j = 0; j < rSize; j++) {
215 for (uint i = 0; i < cSize; i++) {
216 dst[i] -= src[i];
217 }
218 dst += dstStride;
219 src += srcStride;
220 }

231 static inline T* elimDiagBlock(int m, int k, T* A, int lda, T* B) {
232 T* A_row = A;
233 T* B_row = B;
234 for (int i = 0; i < m; i++) {
235 T* C_row = B;
236 for (int j = 0; j <= i; j++) {
237 T& v = A_row[j];
238 for (int q = 0; q < k; q++) {
239 v -= B_row[q] * C_row[q];
240 }
241 C_row += k;
242 }
243 A_row += lda;
244 B_row += k;
245 }
246 return B_row;
247 }

249 static inline T* elimBlock(int m, int n, int k, T* A, int lda, T* B, T* C) {
250 T* A_row = A;
251 T* B_row = B;
252 for (int i = 0; i < m; i++) {
253 T* C_row = C;
254 for (int j = 0; j < n; j++) {
255 T& v = A_row[j];
256 for (int q = 0; q < k; q++) {
257 v -= B_row[q] * C_row[q];
258 }
259 C_row += k;
260 }
261 A_row += lda;
262 B_row += k;
263 }
264 return B_row;
265 }

267 static void eliminateRowChain(const CpuBaseSymElimCtx& elim, const CoalescedBlockMatrixSkel& skel,
268 T* data, int64_t sRel, std::vector<int64_t>& spanToChainOffset) {
269 int64_t s = sRel + elim.spanRowBegin;
270 if (elim.rowPtr[sRel] == elim.rowPtr[sRel + 1]) {
271 return;
272 }
273 int64_t targetLump = skel.spanToLump[s];
274 int64_t targetLumpSize = skel.lumpStart[targetLump + 1] - skel.lumpStart[targetLump];
275 int64_t spanOffsetInLump = skel.spanStart[s] - skel.lumpStart[targetLump];
276 prepareContextForTargetLump(skel, targetLump, spanToChainOffset);
277
278 const int64_t* pChainRowsTillEnd = skel.chainRowsTillEnd.data();
279 const int64_t* pChainRowSpan = skel.chainRowSpan.data();
280 const int64_t* pSpanToChainOffset = spanToChainOffset.data();
281
282 // iterate over chains present in this row
283 for (int64_t i = elim.rowPtr[sRel], iEnd = elim.rowPtr[sRel + 1]; i < iEnd; i++) {
284 int64_t lump = elim.colLump[i];
285 int64_t chainColOrd = elim.chainColOrd[i];
286 BASPACHO_CHECK_GE(chainColOrd,
287 1); // there must be a diagonal block
288
289 int64_t ptrStart = skel.chainColPtr[lump] + chainColOrd;
290 int64_t ptrEnd = skel.chainColPtr[lump + 1];
291 BASPACHO_CHECK_EQ(pChainRowSpan[ptrStart], s);
292
293 int64_t nRowsAbove = pChainRowsTillEnd[ptrStart - 1];
294 int64_t nRowsChain = pChainRowsTillEnd[ptrStart] - nRowsAbove;
295 T* origDataStart = data + skel.chainData[ptrStart];
296 BASPACHO_CHECK_EQ(nRowsChain, skel.spanStart[s + 1] - skel.spanStart[s]);
297 int64_t lumpSize = skel.lumpStart[lump + 1] - skel.lumpStart[lump];
298
299 T* headTargetData = data + spanOffsetInLump + pSpanToChainOffset[pChainRowSpan[ptrStart]];
300 T* nextDataStart =
301 elimDiagBlock(nRowsChain, lumpSize, headTargetData, targetLumpSize, origDataStart);
302
303 // assemble blocks, iterating on chain and below chains
304 int64_t nextRelRow = pChainRowsTillEnd[ptrStart] - nRowsAbove;
305 for (int64_t ptr = ptrStart + 1; ptr < ptrEnd; ptr++) {
306 int64_t s2 = pChainRowSpan[ptr];
307 int64_t relRow = nextRelRow;
308 nextRelRow = pChainRowsTillEnd[ptr] - nRowsAbove;
309 int64_t s2_size = nextRelRow - relRow;
310
311 // incomment below if check is needed
312 // BASPACHO_CHECK(spanToChainOffset[s2] != kInvalid);
313 T* targetData = data + spanOffsetInLump + pSpanToChainOffset[s2];
314
315 nextDataStart = elimBlock(s2_size, nRowsChain, lumpSize, targetData, targetLumpSize,
316 nextDataStart, origDataStart);
317 }
318 }
319 }

321 static void eliminateVerySparseRowChain(const CpuBaseSymElimCtx& elim,
322 const CoalescedBlockMatrixSkel& skel, T* data,
323 int64_t sRel, std::vector<T>& reusableTmpBuf) {
324 int64_t s = sRel + elim.spanRowBegin;
325 if (elim.rowPtr[sRel] == elim.rowPtr[sRel + 1]) {
326 return;
327 }
328 int64_t targetLump = skel.spanToLump[s];
329 int64_t targetLumpSize = skel.lumpStart[targetLump + 1] - skel.lumpStart[targetLump];
330 int64_t spanOffsetInLump = skel.spanStart[s] - skel.lumpStart[targetLump];
331 int64_t bisectStart = skel.chainColPtr[targetLump];
332 int64_t bisectEnd = skel.chainColPtr[targetLump + 1];
333
334 // iterate over chains present in this row
335 for (int64_t i = elim.rowPtr[sRel], iEnd = elim.rowPtr[sRel + 1]; i < iEnd; i++) {
336 int64_t lump = elim.colLump[i];
337 int64_t chainColOrd = elim.chainColOrd[i];
338 BASPACHO_CHECK_GE(chainColOrd,
339 1); // there must be a diagonal block
340
341 int64_t ptrStart = skel.chainColPtr[lump] + chainColOrd;
342 int64_t ptrEnd = skel.chainColPtr[lump + 1];
343 BASPACHO_CHECK_EQ(skel.chainRowSpan[ptrStart], s);
344
345 int64_t nRowsAbove = skel.chainRowsTillEnd[ptrStart - 1];
346 int64_t nRowsChain = skel.chainRowsTillEnd[ptrStart] - nRowsAbove;
347 int64_t nRowsOnward = skel.chainRowsTillEnd[ptrEnd - 1];
348 int64_t dataOffset = skel.chainData[ptrStart];
349 BASPACHO_CHECK_EQ(nRowsChain, skel.spanStart[s + 1] - skel.spanStart[s]);
350 int64_t lumpSize = skel.lumpStart[lump + 1] - skel.lumpStart[lump];
351
352 Eigen::Map<MatRMaj<T>> chainSubMat(data + dataOffset, nRowsChain, lumpSize);
353 Eigen::Map<MatRMaj<T>> chainOnwardSubMat(data + dataOffset, nRowsOnward, lumpSize);
354
355 reusableTmpBuf.resize(nRowsOnward * nRowsChain);
356 Eigen::Map<MatRMaj<T>> prod(reusableTmpBuf.data(), nRowsOnward, nRowsChain);
357 prod.noalias() = chainOnwardSubMat * chainSubMat.transpose();
358
359 // assemble blocks, iterating on chain and below chains
360 for (int64_t ptr = ptrStart; ptr < ptrEnd; ptr++) {
361 int64_t s2 = skel.chainRowSpan[ptr];
362 int64_t relRow = skel.chainRowsTillEnd[ptr - 1] - nRowsAbove;
363 int64_t s2_size = skel.chainRowsTillEnd[ptr] - nRowsAbove - relRow;
364
365 int64_t pos = bisect(skel.chainRowSpan.data() + bisectStart, bisectEnd - bisectStart, s2);
366 int64_t chainOffset = skel.chainData[bisectStart + pos];
367 T* targetData = data + spanOffsetInLump + chainOffset;
368
369 stridedMatSub(targetData, targetLumpSize, prod.data() + nRowsChain * relRow, nRowsChain,
370 s2_size, nRowsChain);
371 }
372 }
373 }

381template <typename T>
382struct CpuBaseSolveCtx : SolveCtx<T> {
383 CpuBaseSolveCtx(const CpuBaseSymbolicCtx& sym, int nRHS)
384 : sym(sym), nRHS(nRHS), tmpBuf(sym.skel.order() * nRHS) {}
385 virtual ~CpuBaseSolveCtx() override {}

387 virtual void symm(const T* data, int64_t offM, int64_t n, const T* C, int64_t offC, int64_t ldc,
388 T* D, int64_t ldd, T alpha) override {
389 auto timer = sym.symmStat.instance();
390 Eigen::Map<const MatRMaj<T>> matA(data + offM, n, n);
391 OuterStridedCMajMatK<T> matC(C + offC, n, nRHS, OuterStride(ldc));
392 OuterStridedCMajMatM<T> matD(D + offC, n, nRHS, OuterStride(ldd));
393
394 matD.noalias() += alpha * (MatRMaj<T>(matA.template selfadjointView<Eigen::Lower>()) * matC);
395 }
396
397 virtual void solveL(const T* data, int64_t offM, int64_t n, T* C, int64_t offC,
398 int64_t ldc) override {
399 auto timer = sym.solveLStat.instance();
400 Eigen::Map<const MatRMaj<T>> matA(data + offM, n, n);
401 OuterStridedCMajMatM<T> matC(C + offC, n, nRHS, OuterStride(ldc));
402 matA.template triangularView<Eigen::Lower>().solveInPlace(matC);
403 }
404
405 virtual void gemv(const T* data, int64_t offM, int64_t nRows, int64_t nCols, const T* A,
406 int64_t offA, int64_t lda, T alpha) override {
407 auto timer = sym.solveGemvStat.instance();
408 Eigen::Map<const MatRMaj<T>> matM(data + offM, nRows, nCols);
409 OuterStridedCMajMatK<T> matA(A + offA, nCols, nRHS, OuterStride(lda));
410 Eigen::Map<MatRMaj<T>> matC(tmpBuf.data(), nRows, nRHS);
411 matC.noalias() = alpha * (matM * matA);
412 }
413
414 virtual void solveLt(const T* data, int64_t offM, int64_t n, T* C, int64_t offC,
415 int64_t ldc) override {
416 auto timer = sym.solveLtStat.instance();
417 Eigen::Map<const MatRMaj<T>> matA(data + offM, n, n);
418 OuterStridedCMajMatM<T> matC(C + offC, n, nRHS, OuterStride(ldc));
419 matA.template triangularView<Eigen::Lower>().adjoint().solveInPlace(matC);
420 }
421
422 virtual void gemvT(const T* data, int64_t offM, int64_t nRows, int64_t nCols, T* A, int64_t offA,
423 int64_t lda, T alpha) override {
424 auto timer = sym.solveGemvTStat.instance();
425 Eigen::Map<const MatRMaj<T>> matM(data + offM, nRows, nCols);
426 OuterStridedCMajMatM<T> matA(A + offA, nCols, nRHS, OuterStride(lda));
427 Eigen::Map<const MatRMaj<T>> matC(tmpBuf.data(), nRows, nRHS);
428 matA.noalias() += alpha * (matM.transpose() * matC);
429 }

431 const CpuBaseSymbolicCtx& sym;
432 int64_t nRHS;
433 std::vector<T> tmpBuf;

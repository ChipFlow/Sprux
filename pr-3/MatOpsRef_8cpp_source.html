64 virtual void doElimination(const SymElimCtx& elimData, T* data, int64_t lumpsBegin,
65 int64_t lumpsEnd) override {
66 const CpuBaseSymElimCtx* pElim = dynamic_cast<const CpuBaseSymElimCtx*>(&elimData);
67 BASPACHO_CHECK_NOTNULL(pElim);
68 const CpuBaseSymElimCtx& elim = *pElim;
69 auto timer = elim.elimStat.instance();
70 const CoalescedBlockMatrixSkel& skel = sym.skel;
71
72 for (int64_t l = lumpsBegin; l < lumpsEnd; l++) {
73 factorLump(skel, data, l);
74 }
75
76 int64_t numElimRows = elim.rowPtr.size() - 1;
77 int64_t numSpans = skel.spanStart.size() - 1;
78 std::vector<int64_t> spanToChainOffset(numSpans);
79 for (int64_t sRel = 0UL; sRel < numElimRows; sRel++) {
80 eliminateRowChain(elim, skel, data, sRel, spanToChainOffset);
81 }
82 }

84 // This is an alternative of doElimination, which is however written in such
85 // a way to mock the sparse elimination as done on the GPU. This is in order
86 // to test the logic here, and then create a kernel doing something similar
87 void doEliminationMockSparse(const SymElimCtx& elimData, T* data, int64_t lumpsBegin,
88 int64_t lumpsEnd) {
89 const CpuBaseSymElimCtx* pElim = dynamic_cast<const CpuBaseSymElimCtx*>(&elimData);
90 BASPACHO_CHECK_NOTNULL(pElim);
91 const CpuBaseSymElimCtx& elim = *pElim;
92 auto timer = elim.elimStat.instance();
93 const CoalescedBlockMatrixSkel& skel = sym.skel;
94
95 for (int64_t l = lumpsBegin; l < lumpsEnd; l++) {
96 factorLump(skel, data, l);
97 }
98
99 for (int64_t l = lumpsBegin; l < lumpsEnd; l++) {
100 int64_t startPtr = skel.chainColPtr[l] + 1; // skip diag block
101 int64_t endPtr = skel.chainColPtr[l + 1];
102 int64_t lColSize = skel.lumpStart[l + 1] - skel.lumpStart[l];
103
104 for (int64_t i = startPtr; i < endPtr; i++) {
105 int64_t si = skel.chainRowSpan[i];
106 int64_t siSize = skel.spanStart[si + 1] - skel.spanStart[si];
107 int64_t siDataPtr = skel.chainData[i];
108 Eigen::Map<MatRMaj<T>> ilBlock(data + siDataPtr, siSize, lColSize);
109
110 int64_t targetLump = skel.spanToLump[si];
111 int64_t targetSpanOffsetInLump = skel.spanOffsetInLump[si];
112 int64_t targetStartPtr = skel.chainColPtr[targetLump]; // skip diag block
113 int64_t targetEndPtr = skel.chainColPtr[targetLump + 1];
114 int64_t targetLumpSize = skel.lumpStart[targetLump + 1] - skel.lumpStart[targetLump];
115
116 for (int64_t j = i; j < endPtr; j++) {
117 int64_t sj = skel.chainRowSpan[j];
118 int64_t sjSize = skel.spanStart[sj + 1] - skel.spanStart[sj];
119 int64_t sjDataPtr = skel.chainData[j];
120
121 Eigen::Map<MatRMaj<T>> jlBlock(data + sjDataPtr, sjSize, lColSize);
122
123 uint64_t pos =
124 bisect(skel.chainRowSpan.data() + targetStartPtr, targetEndPtr - targetStartPtr, sj);
125 BASPACHO_CHECK_EQ(skel.chainRowSpan[targetStartPtr + pos], sj);
126 int64_t jiDataPtr = skel.chainData[targetStartPtr + pos];
127 OuterStridedMatM<T> jiBlock(data + jiDataPtr + targetSpanOffsetInLump, sjSize, siSize,
128 OuterStride(targetLumpSize));
129 jiBlock.noalias() -= jlBlock * ilBlock.transpose();
130 }
131 }
132 }

144 virtual void assemble(T* data, int64_t rectRowBegin,
145 int64_t dstStride, //
146 int64_t srcColDataOffset, int64_t srcRectWidth, int64_t numBlockRows,
147 int64_t numBlockCols) override {
148 auto timer = sym.asmblStat.instance(sizeof(T), numBlockRows, numBlockCols);
149 const CoalescedBlockMatrixSkel& skel = sym.skel;
150 const int64_t* chainRowsTillEnd = skel.chainRowsTillEnd.data() + srcColDataOffset;
151 const int64_t* pToSpan = skel.chainRowSpan.data() + srcColDataOffset;
152 const int64_t* pSpanToChainOffset = spanToChainOffset.data();
153 const int64_t* pSpanOffsetInLump = skel.spanOffsetInLump.data();
154
155 const T* matRectPtr = tempBuffer.data();
156
157 for (int64_t r = 0; r < numBlockRows; r++) {
158 int64_t rBegin = chainRowsTillEnd[r - 1] - rectRowBegin;
159 int64_t rSize = chainRowsTillEnd[r] - rBegin - rectRowBegin;
160 int64_t rParam = pToSpan[r];
161 int64_t rOffset = pSpanToChainOffset[rParam];
162 const T* matRowPtr = matRectPtr + rBegin * srcRectWidth;
163
164 int64_t cEnd = std::min(numBlockCols, r + 1);
165 for (int64_t c = 0; c < cEnd; c++) {
166 int64_t cStart = chainRowsTillEnd[c - 1] - rectRowBegin;
167 int64_t cSize = chainRowsTillEnd[c] - cStart - rectRowBegin;
168 int64_t offset = rOffset + pSpanOffsetInLump[pToSpan[c]];
169
170 T* dst = data + offset;
171 const T* src = matRowPtr + cStart;
172 stridedMatSub(dst, dstStride, src, srcRectWidth, rSize, cSize);
173 }
174 }
175 }

194 virtual void sparseElimSolveL(const SymElimCtx& elimData, const T* data, int64_t lumpsBegin,
195 int64_t lumpsEnd, T* C, int64_t ldc) override {
196 auto timer = sym.solveSparseLStat.instance();
197 const CpuBaseSymElimCtx* pElim = dynamic_cast<const CpuBaseSymElimCtx*>(&elimData);
198 BASPACHO_CHECK_NOTNULL(pElim);
199 const CpuBaseSymElimCtx& elim = *pElim;
200 const CoalescedBlockMatrixSkel& skel = sym.skel;
201
202 for (int64_t lump = lumpsBegin; lump < lumpsEnd; lump++) {
203 int64_t lumpStart = skel.lumpStart[lump];
204 int64_t lumpSize = skel.lumpStart[lump + 1] - lumpStart;
205 int64_t colStart = skel.chainColPtr[lump];
206 int64_t diagDataPtr = skel.chainData[colStart];
207
208 // in-place lower diag cholesky dec on diagonal block
209 Eigen::Map<const MatRMaj<T>> diagBlock(data + diagDataPtr, lumpSize, lumpSize);
210 OuterStridedCMajMatM<T> matC(C + lumpStart, lumpSize, nRHS, OuterStride(ldc));
211 diagBlock.template triangularView<Eigen::Lower>().solveInPlace(matC);
212
213#if 0
214 // alternative to the below, keep here for reference
215 int64_t colEnd = skel.chainColPtr[lump + 1];
216 for (int64_t colPtr = colStart + 1; colPtr < colEnd; colPtr++) {
217 int64_t rowSpan = skel.chainRowSpan[colPtr];
218 int64_t rowSpanStart = skel.spanStart[rowSpan];
219 int64_t rowSpanSize =
220 skel.spanStart[rowSpan + 1] - rowSpanStart;
221 int64_t blockPtr = skel.chainData[colPtr];
222 Eigen::Map<const MatRMaj<T>> block(data + blockPtr, rowSpanSize,
223 lumpSize);
224 OuterStridedCMajMatM<T> matQ(C + rowSpanStart, rowSpanSize,
225 nRHS, OuterStride(ldc));
226 matQ.noalias() -= block * matC;
227 }
228#endif
229 }
230
231 // per-row iterations (like during sparse elimination, using elim data):
232 // in this way the outer loop can be parallelized
233 int64_t numElimRows = elim.rowPtr.size() - 1;
234 for (int64_t sRel = 0UL; sRel < numElimRows; sRel++) {
235 int64_t rowSpan = sRel + elim.spanRowBegin;
236 int64_t rowSpanStart = skel.spanStart[rowSpan];
237 int64_t rowSpanSize = skel.spanStart[rowSpan + 1] - rowSpanStart;
238 OuterStridedCMajMatM<T> matQ(C + rowSpanStart, rowSpanSize, nRHS, OuterStride(ldc));
239
240 for (int64_t i = elim.rowPtr[sRel], iEnd = elim.rowPtr[sRel + 1]; i < iEnd; i++) {
241 int64_t lump = elim.colLump[i];
242 int64_t lumpStart = skel.lumpStart[lump];
243 int64_t lumpSize = skel.lumpStart[lump + 1] - lumpStart;
244 int64_t chainColOrd = elim.chainColOrd[i];
245 BASPACHO_CHECK_GE(chainColOrd,
246 1); // there must be a diagonal block
247
248 int64_t ptr = skel.chainColPtr[lump] + chainColOrd;
249 BASPACHO_CHECK_EQ(skel.chainRowSpan[ptr], rowSpan);
250 int64_t blockPtr = skel.chainData[ptr];
251
252 Eigen::Map<const MatRMaj<T>> block(data + blockPtr, rowSpanSize, lumpSize);
253 OuterStridedCMajMatM<T> matC(C + lumpStart, lumpSize, nRHS, OuterStride(ldc));
254 matQ.noalias() -= block * matC;
255 }
256 }
257 }

259 virtual void sparseElimSolveLt(const SymElimCtx& /*elimData*/, const T* data, int64_t lumpsBegin,
260 int64_t lumpsEnd, T* C, int64_t ldc) override {
261 auto timer = sym.solveSparseLtStat.instance();
262 const CoalescedBlockMatrixSkel& skel = sym.skel;
263
264 // outer loop can be parallelized
265 for (int64_t lump = lumpsBegin; lump < lumpsEnd; lump++) {
266 int64_t lumpStart = skel.lumpStart[lump];
267 int64_t lumpSize = skel.lumpStart[lump + 1] - lumpStart;
268 int64_t colStart = skel.chainColPtr[lump];
269 int64_t colEnd = skel.chainColPtr[lump + 1];
270 OuterStridedCMajMatM<T> matC(C + lumpStart, lumpSize, nRHS, OuterStride(ldc));
271
272 for (int64_t colPtr = colStart + 1; colPtr < colEnd; colPtr++) {
273 int64_t rowSpan = skel.chainRowSpan[colPtr];
274 int64_t rowSpanStart = skel.spanStart[rowSpan];
275 int64_t rowSpanSize = skel.spanStart[rowSpan + 1] - rowSpanStart;
276 int64_t blockPtr = skel.chainData[colPtr];
277 Eigen::Map<const MatRMaj<T>> block(data + blockPtr, rowSpanSize, lumpSize);
278 OuterStridedCMajMatM<T> matQ(C + rowSpanStart, rowSpanSize, nRHS, OuterStride(ldc));
279 matC.noalias() -= block.transpose() * matQ;
280 }
281
282 int64_t diagDataPtr = skel.chainData[colStart];
283 Eigen::Map<const MatRMaj<T>> diagBlock(data + diagDataPtr, lumpSize, lumpSize);
284 diagBlock.template triangularView<Eigen::Lower>().adjoint().solveInPlace(matC);
285 }
286 }

288 virtual void assembleVec(int64_t chainColPtr, int64_t numColItems, T* C, int64_t ldc) override {
289 auto timer = sym.solveAssVStat.instance();
290 const T* A = tmpBuf.data();
291 const CoalescedBlockMatrixSkel& skel = sym.skel;
292 const int64_t* chainRowsTillEnd = skel.chainRowsTillEnd.data() + chainColPtr;
293 const int64_t* toSpan = skel.chainRowSpan.data() + chainColPtr;
294 int64_t startRow = chainRowsTillEnd[-1];
295 for (int64_t i = 0; i < numColItems; i++) {
296 int64_t rowOffset = chainRowsTillEnd[i - 1] - startRow;
297 int64_t span = toSpan[i];
298 int64_t spanStart = skel.spanStart[span];
299 int64_t spanSize = skel.spanStart[span + 1] - spanStart;
300
301 Eigen::Map<const MatRMaj<T>> matA(A + rowOffset * nRHS, spanSize, nRHS);
302 OuterStridedCMajMatM<T> matC(C + spanStart, spanSize, nRHS, OuterStride(ldc));
303 matC.noalias() += matA;
304 }
305 }

323 matA = matC;
324 }
325 }

17CoalescedBlockMatrixSkel::CoalescedBlockMatrixSkel(const vector<int64_t>& spanStart,
18 const vector<int64_t>& lumpToSpan,
19 const vector<int64_t>& colPtr,
20 const vector<int64_t>& rowInd)
21 : spanStart(spanStart), lumpToSpan(lumpToSpan) {
22 BASPACHO_CHECK_GE(spanStart.size(), lumpToSpan.size());
23 BASPACHO_CHECK_GE((int64_t)lumpToSpan.size(), 1);
24 BASPACHO_CHECK_EQ((int64_t)spanStart.size() - 1, lumpToSpan[lumpToSpan.size() - 1]);
25 BASPACHO_CHECK_EQ(colPtr.size(), lumpToSpan.size());
26 BASPACHO_CHECK(isStrictlyIncreasing(spanStart, 0, spanStart.size()));
27 BASPACHO_CHECK(isStrictlyIncreasing(lumpToSpan, 0, lumpToSpan.size()));
28
29 int64_t totSize = spanStart[spanStart.size() - 1];
30 int64_t numSpans = spanStart.size() - 1;
31 int64_t numLumps = lumpToSpan.size() - 1;
32
33 spanToLump.resize(numSpans + 1);
34 lumpStart.resize(numLumps + 1);
35 for (int64_t l = 0; l < numLumps; l++) {
36 int64_t sBegin = lumpToSpan[l];
37 int64_t sEnd = lumpToSpan[l + 1];
38 lumpStart[l] = spanStart[sBegin];
39 for (int64_t s = sBegin; s < sEnd; s++) {
40 spanToLump[s] = l;
41 }
42 }
43 spanToLump[numSpans] = numLumps;
44 lumpStart[numLumps] = totSize;
45 spanOffsetInLump.resize(numSpans + 1);
46 for (int64_t s = 0; s < numSpans; s++) {
47 spanOffsetInLump[s] = spanStart[s] - lumpStart[spanToLump[s]];
48 }
49 spanOffsetInLump[numSpans] = 0;
50
51 chainColPtr.resize(numLumps + 1);
52 chainRowSpan.clear();
53 chainData.clear();
54
55 boardColPtr.resize(numLumps + 1);
56 boardRowLump.clear();
57 boardChainColOrd.clear();
58 int64_t dataPtr = 0;
59 for (int64_t l = 0; l < numLumps; l++) {
60 int64_t colStart = colPtr[l];
61 int64_t colEnd = colPtr[l + 1];
62 BASPACHO_CHECK(isStrictlyIncreasing(rowInd, colStart, colEnd));
63 int64_t lSpanBegin = lumpToSpan[l];
64 int64_t lSpanEnd = lumpToSpan[l + 1];
65 int64_t lSpanSize = lSpanEnd - lSpanBegin;
66 int64_t lDataSize = lumpStart[l + 1] - lumpStart[l];
67
68 // check the initial section is the set of params from `a`, and
69 // therefore the full diagonal block is contained in the matrix
70 // Column must contain full diagonal block:
71 BASPACHO_CHECK_GE(colEnd - colStart, lSpanSize);
72 // Column data must start at diagonal block:
73 BASPACHO_CHECK_EQ(rowInd[colStart], lSpanBegin);
74 // Column must contain full diagonal block:
75 BASPACHO_CHECK_EQ(rowInd[colStart + lSpanSize - 1], lSpanEnd - 1);
76
77 chainColPtr[l] = chainRowSpan.size();
78 boardColPtr[l] = boardRowLump.size();
79 int64_t currentRowAggreg = kInvalid;
80 int64_t numRowsSkipped = 0;
81 for (int64_t i = colStart; i < colEnd; i++) {
82 int64_t p = rowInd[i];
83 chainRowSpan.push_back(p);
84 chainData.push_back(dataPtr);
85 dataPtr += lDataSize * (spanStart[p + 1] - spanStart[p]);
86 numRowsSkipped += spanStart[p + 1] - spanStart[p];
87 chainRowsTillEnd.push_back(numRowsSkipped);
88
89 int64_t rowAggreg = spanToLump[p];
90 if (rowAggreg != currentRowAggreg) {
91 currentRowAggreg = rowAggreg;
92 boardRowLump.push_back(rowAggreg);
93 boardChainColOrd.push_back(i - colStart);
94 }
95 }
96 boardRowLump.push_back(kInvalid);
97 boardChainColOrd.push_back(colEnd - colStart);
98 }
99 chainColPtr[numLumps] = chainRowSpan.size();
100 boardColPtr[numLumps] = boardRowLump.size();
101 chainData.push_back(dataPtr);
102
103 boardRowPtr.assign(numLumps + 1, 0);
104 for (int64_t l = 0; l < numLumps; l++) {
105 for (int64_t i = boardColPtr[l]; i < boardColPtr[l + 1] - 1; i++) {
106 int64_t rowLump = boardRowLump[i];
107 boardRowPtr[rowLump]++;
108 }
109 }
110 int64_t numBoards = cumSumVec(boardRowPtr);
111 boardColLump.resize(numBoards);
112 boardColOrd.resize(numBoards);
113 for (int64_t l = 0; l < numLumps; l++) {
114 for (int64_t i = boardColPtr[l]; i < boardColPtr[l + 1] - 1; i++) {
115 int64_t rowLump = boardRowLump[i];
116 boardColLump[boardRowPtr[rowLump]] = l;
117 boardColOrd[boardRowPtr[rowLump]] = i - boardColPtr[l];
118 boardRowPtr[rowLump]++;
119 }
120 }
121 rewindVec(boardRowPtr);
122}

124template <typename T>
125void CoalescedBlockMatrixSkel::densify(Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& dense,
126 const T* data, bool fillUpperHalf,
127 int64_t startSpanIndex) const {
128 BASPACHO_CHECK_GE(startSpanIndex, 0);
129 BASPACHO_CHECK_LT(startSpanIndex, (int64_t)spanOffsetInLump.size());
130 BASPACHO_CHECK_EQ(spanOffsetInLump[startSpanIndex], 0);
131 int64_t startLump = spanToLump[startSpanIndex];
132
133 int64_t offset = spanStart[startSpanIndex];
134 int64_t totSize = spanStart[spanStart.size() - 1] - offset;
135 dense.resize(totSize, totSize);
136 dense.setZero();
137
138 for (size_t a = startLump; a < chainColPtr.size() - 1; a++) {
139 int64_t lBegin = lumpStart[a];
140 int64_t lSize = lumpStart[a + 1] - lBegin;
141 int64_t colStart = chainColPtr[a];
142 int64_t colEnd = chainColPtr[a + 1];
143 for (int64_t i = colStart; i < colEnd; i++) {
144 int64_t p = chainRowSpan[i];
145 int64_t pStart = spanStart[p];
146 int64_t pSize = spanStart[p + 1] - pStart;
147 int64_t dataPtr = chainData[i];
148
149 dense.block(pStart - offset, lBegin - offset, pSize, lSize) =
150 Eigen::Map<const MatRMaj<T>>(data + dataPtr, pSize, lSize);
151 }
152 }
153
154 if (fillUpperHalf) {
155 dense.template triangularView<Eigen::StrictlyUpper>() =
156 dense.template triangularView<Eigen::StrictlyLower>().adjoint();
157 }
158}

160template <typename T>
161Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic> CoalescedBlockMatrixSkel::densify(
162 const std::vector<T>& data, bool fillUpperHalf) const {
163 int64_t totData = chainData[chainData.size() - 1];
164 BASPACHO_CHECK_EQ(totData, (int64_t)data.size());
165
166 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic> dense;
167 densify(dense, data.data(), fillUpperHalf);
168
169 return dense;
170}

172template <typename T>
173void CoalescedBlockMatrixSkel::damp(std::vector<T>& data, T alpha, T beta) const {
174 int64_t totData = chainData[chainData.size() - 1];
175 BASPACHO_CHECK_EQ(totData, (int64_t)data.size());
176
177 for (size_t a = 0; a < chainColPtr.size() - 1; a++) {
178 int64_t aStart = lumpStart[a];
179 int64_t aSize = lumpStart[a + 1] - aStart;
180 int64_t colStart = chainColPtr[a];
181 int64_t dataPtr = chainData[colStart];
182
183 Eigen::Map<MatRMaj<T>> block(data.data() + dataPtr, aSize, aSize);
184 block.diagonal() *= (1 + alpha);
185 block.diagonal().array() += beta;
186 }
187}

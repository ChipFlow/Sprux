48// utility to collect timing stats
49template <typename... Args>
50struct OpStat {
51 template <typename SyncOps>
52 struct Instance {
53 Instance() : stat(nullptr) {}
54
55 Instance& operator=(Instance&& that) noexcept {
56 stat = that.stat;
57 start = that.start;
58 stat = nullptr;
59 }
60
61 Instance(Instance&& that) noexcept : stat(that.stat), start(that.start) { that.stat = nullptr; }
62
63 Instance(OpStat* stat, const Args&... args)
64 : stat(stat), start(stat ? hrc::now() : TimePoint()), argsTuple(args...) {}
65
66 template <int i, int j, int... ns>
67 inline typename std::enable_if<(i < j)>::type invokeCallback(double runTime) {
68 invokeCallback<i + 1, j, ns..., i>(runTime);
69 }
70
71 template <int i, int j, int... ns>
72 inline typename std::enable_if<(i == j)>::type invokeCallback(double runTime) {
73 stat->callBack(runTime, std::get<ns>(argsTuple)...);
74 }
75
76 ~Instance() {
77 if (!stat) {
78 return;
79 }
80 SyncOps::sync();
81 double runTime = tdelta(hrc::now() - start).count();
82 stat->numRuns++;
83 stat->lastTime = runTime;
84 stat->maxTime = std::max(stat->maxTime, runTime);
85 stat->totTime += runTime;
86 if (stat->callBack) {
87 invokeCallback<0, sizeof...(Args)>(runTime);
88 }
89 }
90
91 OpStat* stat;
92 TimePoint start;
93 std::tuple<Args...> argsTuple;
94 };
95
96 std::string toString() const {
97 std::stringstream ss;
98 ss << "#=" << numRuns << ", time=" << secondsToString(totTime)
99 << ", last=" << secondsToString(lastTime) << ", max=" << secondsToString(maxTime);
100 return ss.str();
101 }
102
103 void reset() {
104 numRuns = 0;
105 totTime = 0;
106 maxTime = 0;
107 lastTime = 0;
108 }
109
110 template <typename SyncOps = DefaultSyncOps>
111 Instance<SyncOps> instance(const Args&... args) {
112 return enabled ? Instance<SyncOps>(this, args...) : Instance<SyncOps>();
113 }
114
115 bool enabled = true;
116 int64_t numRuns = 0;
117 double totTime = 0;
118 double maxTime = 0;
119 double lastTime = 0;
120 std::function<void(double, const Args&... args)> callBack;
121};

123// utility, check if vector range is strictly increasion
124template <typename T>
125bool isStrictlyIncreasing(const std::vector<T>& v, std::size_t begin, std::size_t e) {
126 std::size_t i = begin + 1;
127 while (i < e && (v[i] > v[i - 1])) {
128 i++;
129 }
130 return i == e;
131}

155inline int64_t bisect(const int64_t* array, int64_t size, int64_t needle) {
156 int64_t a = 0, b = size;
157 while (b - a > 1) {
158 int64_t m = (a + b) / 2;
159 if (needle >= array[m]) {
160 a = m;
161 } else {
162 b = m;
163 }
164 }
165 return a;
166}

168// appends all elements x shifted by a given amount (x + shift)
169template <class ForwardIt>
170inline void shiftConcat(std::vector<int64_t>& target, int64_t shift, ForwardIt first,
171 ForwardIt last) {
172 while (first != last) {
173 target.push_back(shift + *first++);

177// apply permutation on the left: it[perm[i]] = w[i]
178template <typename T, typename ForwardIt>
179void leftPermute(ForwardIt it, const std::vector<int64_t>& perm, const std::vector<T>& w) {
180 for (size_t i = 0; i < perm.size(); i++) {
181 *(it + perm[i]) = w[i];
182 }

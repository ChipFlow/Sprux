23/*
24 Notation for (symmetric) block matrix with coalesced columns:
25 Linear data:
26 * a `span` is the basic grouping of data (at creation)
27 * an `lump` is formed by a few consecutive params
28 Block data:
29 * a `block` is `span rows` x `span cols`
30 * a `chain` is `span rows` x `lump cols`
31 * a `board` is (non-empty) and formed by:
32 `all the spans belonging to an lump of rows` x `lump cols`
33
34 Note that numeric data in a column of chains are a row-major
35 matrix. In this way we can refer to the chain sub-matrix,
36 or to the whole set of columns as the chain data are consecutive.
37*/
38struct CoalescedBlockMatrixSkel {
39 CoalescedBlockMatrixSkel(const std::vector<int64_t>& spanStart,
40 const std::vector<int64_t>& lumpToSpan,
41 const std::vector<int64_t>& colPtr, const std::vector<int64_t>& rowInd);
42
43 /* densify the data pointed to by `data`.
44 by default only lower half is filled, unless `fillUpperHalf` is true
45 if startSpanIndex is set it must be on supernode boundary, and then only the
46 bottom right corner will be returned.
47 */
48 template <typename T>
49 void densify(Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& dense, const T* data,
50 bool fillUpperHalf = false, int64_t startSpanIndex = 0) const;
51
52 /* convenience overload */
53 template <typename T>

55 bool fillUpperHalf = false) const;

28void EliminationTree::buildTree() {
29 int64_t ord = ss.order();
30 parent.assign(ord, -1);
31
32 nodeSize = paramSize;
33 nodeRows.assign(ord, 0);
34 nodeRowBlocks.assign(ord, 0);
35 vector<int64_t> tags(ord);
36
37 // skeleton of the algo to iterate on fillup's nodes is from Eigen's
38 // `SimplicialCholesky_impl.h` (by Gael Guennebaud),
39 // in turn from LDL by Timothy A. Davis.
40 perColNodes.resize(ord);
41 for (int64_t k = 0; k < ord; ++k) {
42 /* L(k,:) pattern: all nodes reachable in etree from nz in A(0:k-1,k) */
43 parent[k] = -1; /* parent of k is not yet known */
44 tags[k] = k; /* mark node k as visited */
45 /* L(k,k) is nonzero */
46
47 int64_t start = ss.ptrs[k];
48 int64_t end = ss.ptrs[k + 1];
49 for (int64_t q = start; q < end; q++) {
50 int64_t i = ss.inds[q];
51 if (i >= k) {
52 continue;
53 }
54 /* follow path from i to root of etree, stop at flagged node */
55 for (; tags[i] != k; i = parent[i]) {
56 /* find parent of i if not yet determined */
57 if (parent[i] == -1) {
58 parent[i] = k;
59 }
60
61 tags[i] = k;
62 /* L(k,i) is nonzero */
63 nodeRows[i] += paramSize[k];
64 nodeRowBlocks[i]++;
65 perColNodes[i].push_back(k);
66 }
67 }
68 }
69
70 sygeCosts.resize(ord);
71 asmblCosts.resize(ord);
72 perRowNodeStats.resize(ord);
73 for (int64_t col = 0; col < (int64_t)perColNodes.size(); col++) {
74 auto& c = perColNodes[col];
75 c.push_back(col);
76 sort(c.begin(), c.end());
77
78 int64_t skippedRows = 0;
79 int64_t skippedBlocks = 0;
80 Eigen::Vector2d sygeC{0, 0}, asmblC{0, 0};
81 for (int64_t i = c.size() - 1; i >= 0; i--) {
82 int64_t row = c[i];
83
84 sygeC += compMod.sygeLinEst(skippedRows + paramSize[row], paramSize[row]);
85 asmblC += compMod.asmblLinEst(skippedBlocks + 1);
86
87 perRowNodeStats[row].emplace_back(col, 1, paramSize[row], skippedBlocks, skippedRows);
88
89 skippedRows += paramSize[row];
90 skippedBlocks++;
91 }
92 sygeCosts[col] = sygeC;
93 asmblCosts[col] = asmblC;
94 }
95}

97// Compute node heights:
98// depending on the structure of the matrix, if we have many small leaf
99// nodes (height = 0) we can eliminate them with a sparse elimination
100// step that skips the node-merge adding fill in. It's then possible
101// to eliminate progressively nodes with height=1, 2, etc till when
102// this strategy is no longer effective and we're better off with
103// merged nodes and blas operations.
104// In order to do so we sort nodes according to height.
105// TODO: consider allowing some initial merge of very small nodes?
106void EliminationTree::computeNodeHeights(const vector<int64_t>& noCrossPoints) {
107 int64_t ord = ss.order();
108
109 unmergedHeightNode.resize(ord);
110 forbidMerge.assign(ord, false);
111
112 vector<int64_t> height(ord, 0);
113 for (size_t rangeIndex = 0; rangeIndex < noCrossPoints.size() + 1; rangeIndex++) {
114 int64_t rangeStart = rangeIndex == 0 ? 0 : noCrossPoints[rangeIndex - 1];
115 int64_t rangeEnd = rangeIndex < noCrossPoints.size() ? noCrossPoints[rangeIndex] : ord;
116
117 for (int64_t k = rangeStart; k < rangeEnd; k++) {
118 unmergedHeightNode[k] = make_tuple(height[k], nodeSize[k], k);
119
120 int64_t par = parent[k];
121 if (par == -1) {
122 continue;
123 }
124 if (par >= rangeEnd) {
125 forbidMerge[k] = true;
126 }
127 height[par] = max(height[par], height[k] + 1);
128 }
129 sort(unmergedHeightNode.begin() + rangeStart, unmergedHeightNode.begin() + rangeEnd);
130 }
131}

136void EliminationTree::computeSparseElimRanges(const vector<int64_t>& noCrossPoints) {
137 int64_t ord = ss.order();
138 sparseElimRanges.push_back(0);
139
140 for (size_t rangeIndex = 0; rangeIndex < noCrossPoints.size() + 1; rangeIndex++) {
141 int64_t rangeStart = rangeIndex == 0 ? 0 : noCrossPoints[rangeIndex - 1];
142 int64_t rangeEnd = rangeIndex < noCrossPoints.size() ? noCrossPoints[rangeIndex] : ord;
143
144 int64_t k0 = rangeStart;
145 while (k0 < rangeEnd) {
146 int64_t k1 = k0;
147 int64_t mergeHeight = get<0>(unmergedHeightNode[k0]);
148
149 int64_t numEasyMerge = 0;
150 while (k1 < rangeEnd && get<0>(unmergedHeightNode[k1]) == mergeHeight &&
151 get<1>(unmergedHeightNode[k1]) <= maxSparseElimNodeSize) {
152 int64_t p = parent[k1];
153 if (p >= 0) {
154 double fillAfterMerge = ((double)nodeRows[k1]) / (nodeRows[p] + nodeSize[p]);
155 if (fillAfterMerge > 0.8) {
156 numEasyMerge++;
157 }
158 }
159 k1++;
160 }
161
162 // skip and stop searching if 1. too small, or 2. most nodes are easily merged
163 if ((k1 - k0) < minNumSparseElimNodes || (k1 - k0) < numEasyMerge * 3) {
164 break;
165 }
166
167 for (int64_t k = k0; k < k1; k++) {
168 forbidMerge[get<2>(unmergedHeightNode[k])] = true;
169 }
170 sparseElimRanges.push_back(k1);
171 k0 = k1;
172 }
173 if (k0 < rangeEnd) {
174 break;
175 }
176 }
177 if (sparseElimRanges.size() == 1) {
178 sparseElimRanges.pop_back();
179 }
180}

182void EliminationTree::computeMerges() {
183 int64_t ord = ss.order();
184 numMergedNodes.assign(ord, 1);
185 mergeWith.assign(ord, -1);
186 numMerges = 0;
187
188 // score used to select which child/parent pair we will try to merge first
189 auto pickUpScore = [&](int64_t k, int64_t p) -> double {
190 return ((double)nodeRows[k]) / (nodeRows[p] + nodeSize[p]);
191 };
192
193 priority_queue<tuple<double, int64_t, int64_t>> mergeCandidates;
194 for (int64_t k = ord - 1; k >= 0; k--) {
195 if (forbidMerge[k]) {
196 continue;
197 }
198 int64_t p = parent[k];
199 if (p == -1) {
200 continue;
201 }
202 mergeCandidates.emplace(pickUpScore(k, p), k, p);
203 }
204
205 while (!mergeCandidates.empty()) {
206 auto wasFillAfterMergeKP = mergeCandidates.top();
207 int64_t k = std::get<1>(wasFillAfterMergeKP);
208 int64_t p = std::get<2>(wasFillAfterMergeKP);
209 mergeCandidates.pop();
210
211 auto oldP = p;
212 BASPACHO_CHECK_LT(p, (int64_t)mergeWith.size());
213 while (mergeWith[p] != -1) {
214 p = mergeWith[p];
215 BASPACHO_CHECK_LT(p, (int64_t)mergeWith.size());
216 }
217
218 // parent was merged? value changed, re-prioritize
219 if (oldP != p) {
220 mergeCandidates.emplace(pickUpScore(k, p), k, p);
221 continue;
222 }
223
224 double sk = nodeSize[k], rk = nodeRows[k], sp = nodeSize[p], rp = nodeRows[p], sm = sp + sk;
225
226 // To decide if we're merging the nodes, we compare the estimated runtimes of the (factor)
227 // operations related to the individual unmerged nodes with the runtime of the operations
228 // in the merged node. This isn't 100% accurate because some children elimination operations
229 // might also merge and this is not accounted, but the current estimate should be rather
230 // accurate.
231 double tk = compMod.potrfEst(sk) + compMod.trsmEst(sk, rk) + //
232 sygeCosts[k][0] + sygeCosts[k][1] * sk + //
233 asmblCosts[k][0] + asmblCosts[k][1] * numMergedNodes[k];
234 double tp = compMod.potrfEst(sp) + compMod.trsmEst(sp, rp) + //
235 sygeCosts[p][0] + sygeCosts[p][1] * sp + //
236 asmblCosts[p][0] + asmblCosts[p][1] * numMergedNodes[p];
237 double tm = compMod.potrfEst(sm) + compMod.trsmEst(sm, rp) + //
238 sygeCosts[p][0] + sygeCosts[p][1] * sm + //
239 asmblCosts[p][0] + asmblCosts[p][1] * (numMergedNodes[k] + numMergedNodes[p]);
240 bool willMerge = tm < tk + tp;
241
242 if (willMerge) {
243 int64_t prevNodeSize = nodeSize[p];
244 int64_t prevNumMergedNodes = numMergedNodes[p];
245 mergeWith[k] = p;
246 nodeSize[p] += nodeSize[k];
247 numMergedNodes[p] += numMergedNodes[k];
248 numMerges++;
249
250 auto& kRD = perRowNodeStats[k];
251 auto& pRD = perRowNodeStats[p];
252 for (size_t ik = 0, ip = 0; ik < kRD.size() || ip < pRD.size(); /* */) {
253 if ((ip >= pRD.size()) || (ik < kRD.size() && kRD[ik].colIdx < pRD[ip].colIdx)) {
254 if (kRD[ik].colIdx != k) {
255 tmpRowStats.push_back(kRD[ik]);
256 }
257 ik++;
258 } else if ((ik >= kRD.size()) || (ip < pRD.size() && kRD[ik].colIdx > pRD[ip].colIdx)) {
259 if (pRD[ip].colIdx != p) {
260 tmpRowStats.push_back(pRD[ip]);
261 }
262 ip++;
263 } else {
264 int64_t c = pRD[ip].colIdx;
265 auto& sygeC = sygeCosts[c];
266 auto& asmblC = asmblCosts[c];
267 sygeC -= compMod.sygeLinEst(kRD[ik].rowsDown + kRD[ik].rows, kRD[ik].rows);
268 asmblC -= compMod.asmblLinEst(kRD[ik].rBlocksDown + kRD[ik].rBlocks);
269 sygeC -= compMod.sygeLinEst(pRD[ip].rowsDown + pRD[ip].rows, pRD[ip].rows);
270 asmblC -= compMod.asmblLinEst(pRD[ip].rBlocksDown + pRD[ip].rBlocks);
271 sygeC += compMod.sygeLinEst(pRD[ip].rowsDown + (kRD[ik].rows + pRD[ip].rows),
272 (kRD[ik].rows + pRD[ip].rows));
273 asmblC += compMod.asmblLinEst(pRD[ip].rBlocksDown + (kRD[ik].rBlocks + pRD[ip].rBlocks));
274
275 tmpRowStats.emplace_back(c, kRD[ik].rBlocks + pRD[ip].rBlocks,
276 kRD[ik].rows + pRD[ip].rows, pRD[ip].rBlocksDown,
277 pRD[ip].rowsDown);
278 ik++;
279 ip++;
280 }
281 }
282 auto& sygeC = sygeCosts[p];
283 auto& asmblC = asmblCosts[p];
284 sygeC -= compMod.sygeLinEst(nodeRows[p] + prevNodeSize, prevNodeSize);
285 asmblC -= compMod.asmblLinEst(nodeRowBlocks[p] + prevNumMergedNodes);
286 sygeC += compMod.sygeLinEst(nodeRows[p] + nodeSize[p], nodeSize[p]);
287 asmblC += compMod.asmblLinEst(nodeRowBlocks[p] + numMergedNodes[p]);
288 tmpRowStats.emplace_back(p, numMergedNodes[p], nodeSize[p], nodeRowBlocks[p], nodeRows[p]);
289 swap(perRowNodeStats[p], tmpRowStats);
290 tmpRowStats.clear();
291 } // willmerge
292 }

295// collapse pointer to parent, make parent become root ancestor
296void EliminationTree::collapseMergePointers() {
297 int64_t ord = ss.order();
298
299 for (int64_t k = ord - 1; k >= 0; k--) {
300 int64_t p = mergeWith[k];
301 if (p == -1) {
302 continue;
303 }
304 int64_t a = mergeWith[p];
305 if (a != -1) {
306 mergeWith[k] = a;
307 }
308 }

311void EliminationTree::processTree(bool detectSparseElimRanges, const vector<int64_t>& noCrossPoints,
312 bool findOnlyElims) {
313 int64_t ord = ss.order();
314
315 computeNodeHeights(noCrossPoints);
316
317 // Compute the sparse elimination ranges (after permutation is applied),
318 // and set a flag to forbid merge of nodes which will be sparse-eliminated
319 if (detectSparseElimRanges) {
320 computeSparseElimRanges(noCrossPoints);
321 }
322
323 if (findOnlyElims) {
324 mergeWith.assign(ord, -1);
325 numMergedNodes.assign(ord, 1);
326 numMerges = 0;
327 } else {
328 computeMerges();
329 collapseMergePointers();
330 }
331
332 // We create `lumpStart` and `lumpToSpan` arrays for the
333 // permuted aggregated parameters.
334 // We also set an `unpermutedRootSpanToLump` to register
335 // the lumpIndex of an unpermuted root node.
336 int64_t numLumps = ord - numMerges, lumpIndex = 0;
337 lumpStart.resize(numLumps + 1); // permuted
338 lumpToSpan.resize(numLumps + 1); // permuted
339 vector<int64_t> unpermutedRootSpanToLump(ord, -1);
340
341 for (int64_t i = 0; i < ord; i++) {
342 int64_t k = std::get<2>(unmergedHeightNode[i]);
343 if (mergeWith[k] != -1) {
344 continue;
345 }
346 unpermutedRootSpanToLump[k] = lumpIndex;
347 lumpStart[lumpIndex] = nodeSize[k];
348 lumpToSpan[lumpIndex] = numMergedNodes[k];
349 lumpIndex++;
350 }
351 BASPACHO_CHECK_EQ(lumpIndex, numLumps);
352
353 cumSumVec(lumpStart);
354 cumSumVec(lumpToSpan);
355
356 // we compute the (inverse) permutation: for each span we have a pointer
357 // to the beginning of the lump, it is advanced to get the index the span
358 // is mapped to (the lumpToSpan pointers are rewinded later)
359 permInverse.resize(ord);
360 for (int64_t i = 0; i < ord; i++) {
361 int64_t p = mergeWith[i];
362 int64_t lumpIndex1 = unpermutedRootSpanToLump[p == -1 ? i : p];
363 permInverse[i] = lumpToSpan[lumpIndex1]++; // advance
364 }
365 rewindVec(lumpToSpan); // restore after advancing
366}

368void EliminationTree::computeAggregateStruct(bool fillOnlyForElims) {
369 int64_t ord = ss.order();
370 int64_t numLumps = ord - numMerges;
371
372 SparseStructure tperm = // lower-half csc
373 ss.symmetricPermutation(permInverse, /* lowerHalf = */ false,
374 /* sortIndices = */ false);
375
376 if (fillOnlyForElims) {
377 for (int64_t e = 0; e < (int64_t)sparseElimRanges.size() - 1; e++) {
378 tperm = tperm.addIndependentEliminationFill(sparseElimRanges[e], sparseElimRanges[e + 1]);
379 }
380 } else {
381 tperm = tperm.addFullEliminationFill();
382 }
383 tperm = tperm.transpose();
384
385 vector<int64_t> tags(ord, -1); // check if row el was added already
386 colStart.push_back(0);
387 for (int64_t a = 0; a < numLumps; a++) {
388 int64_t aStart = lumpToSpan[a];
389 int64_t aEnd = lumpToSpan[a + 1];
390 int64_t pStart = tperm.ptrs[aStart];
391 int64_t pEnd = tperm.ptrs[aEnd];
392 for (int64_t i = pStart; i < pEnd; i++) {
393 int64_t p = tperm.inds[i];
394 if (tags[p] < (int64_t)a) {
395 rowParam.push_back(p); // L(p,a) is set
396 tags[p] = a;
397 }
398 }
399 sort(rowParam.begin() + colStart[colStart.size() - 1], rowParam.end());
400 colStart.push_back(rowParam.size());
401 }
402}

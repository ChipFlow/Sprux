74// utility class to mirror an std::vector on the gpu
75template <typename T>
76struct DevMirror {
77 DevMirror() {}
78 DevMirror(const std::vector<T>& vec) { load(vec); }
79 ~DevMirror() { clear(); }
80 void clear() {
81 if (ptr) {
82 cuCHECK(cudaFree(ptr));
83 ptr = nullptr;
84 allocSize = 0;
85 }
86 }
87 void resizeToAtLeast(size_t size) {
88 if (allocSize < size) {
89 clear();
90 }
91 if (!ptr && size > 0) {
92 cuCHECK(cudaMalloc((void**)&ptr, size * sizeof(T)));
93 CHECK_ALLOCATION(ptr, size * sizeof(T));
94 allocSize = size;
95 }
96 }
97 void load(const std::vector<T>& vec) {
98 if (allocSize < vec.size()) {
99 clear();
100 }
101 if (!vec.size()) {
102 return; // cudaMalloc of size 0 fails
103 }
104 if (!ptr) {
105 cuCHECK(cudaMalloc((void**)&ptr, vec.size() * sizeof(T)));
106 CHECK_ALLOCATION(ptr, vec.size() * sizeof(T));
107 allocSize = vec.size();
108 }
109 cuCHECK(cudaMemcpy(ptr, vec.data(), vec.size() * sizeof(T), cudaMemcpyHostToDevice));
110 }
111 void get(std::vector<T>& vec) const {
112 cuCHECK(cudaMemcpy(vec.data(), ptr, vec.size() * sizeof(T), cudaMemcpyDeviceToHost));
113 }
114 T* ptr = nullptr;
115 size_t allocSize = 0;
116};

119template <typename T>
120struct DevPtrMirror {
121 DevPtrMirror() {}
122 DevPtrMirror(const std::vector<T*>& vec, int64_t offset = 0) { load(vec, offset); }
123 void clear() {

125 cuCHECK(cudaFree(ptr));
126 ptr = nullptr;
127 allocSize = 0;

130 void load(const std::vector<T*>& vec, int64_t offset = 0) {
131 T** vecCopy = (T**)alloca(vec.size() * sizeof(T*));
132 for (size_t i = 0; i < vec.size(); i++) {
133 vecCopy[i] = vec[i] + offset;
134 }
135
136 if (allocSize < vec.size()) {
137 clear();
138 }
139 if (!vec.size()) {
140 return; // cudaMalloc of size 0 fails
141 }

144 CHECK_ALLOCATION(ptr, vec.size() * sizeof(T*));
145 allocSize = vec.size();
146 }
147 cuCHECK(cudaMemcpy(ptr, vecCopy, vec.size() * sizeof(T*), cudaMemcpyHostToDevice));
148 }
149 ~DevPtrMirror() { clear(); }
150 T** ptr = nullptr;
151 size_t allocSize = 0;
152};

56 // `potrf` model: t ~= a + b*n + c*n^2 + d*n^3
57 static double potrfModel(const Eigen::Vector<double, 4>& p, double n) {
58 return p[0] + n * (p[1] + n * (p[2] + n * p[3]));
59 }
60
61 // potrf model: derivative
62 static Eigen::Matrix<double, 1, 4> dPotrfModel(double n) {
63 return {1.0, n, n * n, n * n * n}; //
64 }
65
66 // `trsm` model: t ~= a + b*n + c*n^2 + (d + e*n + f*n^2)*k
67 static double trsmModel(const Eigen::Vector<double, 6>& p, double n, double k) {
68 return p[0] + n * (p[1] + n * p[2]) + k * (p[3] + n * (p[4] + n * p[5]));
69 }
70
71 // `trsm` model: derivative
72 static Eigen::Matrix<double, 1, 6> dTrsmModel(double n, double k) {
73 return {1.0, n, n * n, k, k * n, k * n * n};
74 }
75
76 // `syrk/gemm` model, plain model would be:
77 // t ~= a + b*m + c*n + d*k + e*m*n + f*m*k + g*n*k + h*n*m*k
78 // symmetrized in m,n it becomes (putting u=m+n, v=mn the basis of sym functions):
79 // t ~= a + b*u + c*v + d*k + e*u*k + f*v*k
80 static double sygeModel(const Eigen::Vector<double, 6>& p, double m, double n, double k) {
81 return p[0] + (m + n) * p[1] + (m * n) * p[2] + //
82 k * (p[3] + (m + n) * p[4] + (m * n) * p[5]);
83 }
84
85 // `syrk/gemm` model, derivative
86 static Eigen::Matrix<double, 1, 6> dSygeModel(double m, double n, double k) {
87 return {1.0, (m + n), (m * n), k, k * (m + n), k * (m * n)};
88 }
89
90 // `syrk/gemm` model, as linear function of k
91 static Eigen::Vector2d sygeLinModel(const Eigen::Vector<double, 6>& p, double m, double n) {
92 return {p[0] + (m + n) * p[1] + (m * n) * p[2], //
93 p[3] + (m + n) * p[4] + (m * n) * p[5]};
94 }
95
96 // `asmbl` model: t ~= a + b*br + c*bc + d*br*bc
97 static double asmblModel(const Eigen::Vector<double, 4>& p, double br, double bc) {
98 return p[0] + br * p[1] + bc * p[2] + br * bc * p[3];
99 }
100
101 // `asmbl` model: derivative
102 static Eigen::Matrix<double, 1, 4> dAsmblModel(double br, double bc) {
103 return {1.0, br, bc, br * bc};
104 }
105
106 // `asmbl` model, as linear function of bc
107 static Eigen::Vector2d asmblLinModel(const Eigen::Vector<double, 4>& p, double br) {
108 return {p[0] + br * p[1], p[2] + br * p[3]};
109 }
110
111 // pre-built model collection
112 static const ComputationModel model_OpenBlas_i7_1185g7;
113 static const ComputationModel model_Cuda117_2080Ti;
114};

24Solver::Solver(CoalescedBlockMatrixSkel&& factorSkel_, std::vector<int64_t>&& sparseElimRanges_,
25 std::vector<int64_t>&& permutation_, OpsPtr&& ops_, int64_t canFactorUpTo_)
26 : factorSkel(std::move(factorSkel_)),
27 sparseElimRanges(std::move(sparseElimRanges_)),
28 permutation(std::move(permutation_)),
29 canFactorUpTo(canFactorUpTo_),
30 ops(std::move(ops_)) {
31 if (canFactorUpTo < 0) {
32 canFactorUpTo = factorSkel.numSpans();
33 }
34 symCtx = ops->createSymbolicCtx(factorSkel, permutation);
35 for (int64_t l = 0; l + 1 < (int64_t)sparseElimRanges.size(); l++) {
36 elimCtxs.push_back(symCtx->prepareElimination(sparseElimRanges[l], sparseElimRanges[l + 1]));
37 }
38
39 initElimination();
40}

42template <typename T>
43void Solver::factorLump(NumericCtx<T>& numCtx, T* data, int64_t lump) const {
44 int64_t lumpStart = factorSkel.lumpStart[lump];
45 int64_t lumpSize = factorSkel.lumpStart[lump + 1] - lumpStart;
46 int64_t chainColBegin = factorSkel.chainColPtr[lump];
47 int64_t diagBlockOffset = factorSkel.chainData[chainColBegin];
48
49 // compute lower diag cholesky dec on diagonal block
50 numCtx.potrf(lumpSize, data, diagBlockOffset);
51
52 int64_t boardColBegin = factorSkel.boardColPtr[lump];
53 int64_t boardColEnd = factorSkel.boardColPtr[lump + 1];
54 int64_t belowDiagChainColOrd = factorSkel.boardChainColOrd[boardColBegin + 1];
55 int64_t numColChains = factorSkel.boardChainColOrd[boardColEnd - 1];
56 int64_t belowDiagOffset = factorSkel.chainData[chainColBegin + belowDiagChainColOrd];
57 int64_t numRowsBelowDiag = factorSkel.chainRowsTillEnd[chainColBegin + numColChains - 1] -
58 factorSkel.chainRowsTillEnd[chainColBegin + belowDiagChainColOrd - 1];
59 if (numRowsBelowDiag == 0) {
60 return;
61 }
62
63 numCtx.trsm(lumpSize, numRowsBelowDiag, data, diagBlockOffset, belowDiagOffset);
64}

66template <typename T>
67void Solver::eliminateBoard(NumericCtx<T>& numCtx, T* data, int64_t ptr) const {
68 int64_t origLump = factorSkel.boardColLump[ptr];
69 int64_t boardIndexInCol = factorSkel.boardColOrd[ptr];
70
71 int64_t origLumpSize = factorSkel.lumpStart[origLump + 1] - factorSkel.lumpStart[origLump];
72 int64_t chainColBegin = factorSkel.chainColPtr[origLump];
73
74 int64_t boardColBegin = factorSkel.boardColPtr[origLump];
75 int64_t boardColEnd = factorSkel.boardColPtr[origLump + 1];
76
77 int64_t belowDiagChainColOrd = factorSkel.boardChainColOrd[boardColBegin + boardIndexInCol];
78 int64_t rowDataEnd0 = factorSkel.boardChainColOrd[boardColBegin + boardIndexInCol + 1];
79 int64_t rowDataEnd1 = factorSkel.boardChainColOrd[boardColEnd - 1];
80
81 int64_t belowDiagStart = factorSkel.chainData[chainColBegin + belowDiagChainColOrd];
82 int64_t rectRowBegin = factorSkel.chainRowsTillEnd[chainColBegin + belowDiagChainColOrd - 1];
83 int64_t numRowsSub = factorSkel.chainRowsTillEnd[chainColBegin + rowDataEnd0 - 1] - rectRowBegin;
84 int64_t numRowsFull = factorSkel.chainRowsTillEnd[chainColBegin + rowDataEnd1 - 1] - rectRowBegin;
85
86 numCtx.saveSyrkGemm(numRowsSub, numRowsFull, origLumpSize, data, belowDiagStart);
87
88 int64_t targetLump = factorSkel.boardRowLump[boardColBegin + boardIndexInCol];
89 int64_t targetLumpSize = factorSkel.lumpStart[targetLump + 1] - factorSkel.lumpStart[targetLump];
90 int64_t srcColDataOffset = chainColBegin + belowDiagChainColOrd;
91 int64_t numBlockRows = rowDataEnd1 - belowDiagChainColOrd;
92 int64_t numBlockCols = rowDataEnd0 - belowDiagChainColOrd;
93
94 numCtx.assemble(data, rectRowBegin,
95 targetLumpSize, //
96 srcColDataOffset, //
97 numRowsSub, numBlockRows, numBlockCols);
98}

100int64_t Solver::boardElimTempSize(int64_t lump, int64_t boardIndexInCol) const {
101 int64_t chainColBegin = factorSkel.chainColPtr[lump];
102
103 int64_t boardColBegin = factorSkel.boardColPtr[lump];
104 int64_t boardColEnd = factorSkel.boardColPtr[lump + 1];
105
106 int64_t belowDiagChainColOrd = factorSkel.boardChainColOrd[boardColBegin + boardIndexInCol];
107 int64_t rowDataEnd0 = factorSkel.boardChainColOrd[boardColBegin + boardIndexInCol + 1];
108 int64_t rowDataEnd1 = factorSkel.boardChainColOrd[boardColEnd - 1];
109
110 int64_t rectRowBegin = factorSkel.chainRowsTillEnd[chainColBegin + belowDiagChainColOrd - 1];
111 int64_t numRowsSub = factorSkel.chainRowsTillEnd[chainColBegin + rowDataEnd0 - 1] - rectRowBegin;
112 int64_t numRowsFull = factorSkel.chainRowsTillEnd[chainColBegin + rowDataEnd1 - 1] - rectRowBegin;
113
114 return numRowsSub * numRowsFull;
115}

117void Solver::initElimination() {
118 int64_t denseOpsFromLump = sparseElimRanges.size() ? sparseElimRanges.back() : 0;
119
120 startElimRowPtr.resize(factorSkel.chainColPtr.size() - 1 - denseOpsFromLump);
121 maxElimTempSize = 0;
122 for (int64_t l = denseOpsFromLump; l < (int64_t)factorSkel.chainColPtr.size() - 1; l++) {
123 // iterate over columns having a non-trivial a-block
124 int64_t rPtr0 = factorSkel.boardRowPtr[l];
125 int64_t rEnd0 = factorSkel.boardRowPtr[l + 1];
126 BASPACHO_CHECK_EQ(factorSkel.boardColLump[rEnd0 - 1], l);
127 while (factorSkel.boardColLump[rPtr0] < denseOpsFromLump) {
128 rPtr0++;
129 }
130 BASPACHO_CHECK_LT(rPtr0,
131 rEnd0); // will stop before end as l > denseOpsFromLump
132 startElimRowPtr[l - denseOpsFromLump] = rPtr0;
133
134 for (int64_t rPtr = startElimRowPtr[l - denseOpsFromLump],
135 rEnd = factorSkel.boardRowPtr[l + 1]; //
136 rPtr < rEnd && factorSkel.boardColLump[rPtr] < l; //
137 rPtr++) {
138 int64_t origLump = factorSkel.boardColLump[rPtr];
139 int64_t boardIndexInCol = factorSkel.boardColOrd[rPtr];
140 int64_t boardSNDataStart = factorSkel.boardColPtr[origLump];
141 int64_t boardSNDataEnd = factorSkel.boardColPtr[origLump + 1];
142 BASPACHO_CHECK_LT(boardIndexInCol, boardSNDataEnd - boardSNDataStart);
143 BASPACHO_CHECK_EQ(l, factorSkel.boardRowLump[boardSNDataStart + boardIndexInCol]);
144 maxElimTempSize = max(maxElimTempSize, boardElimTempSize(origLump, boardIndexInCol));
145 }
146 }
147}

164template <typename T>
165void Solver::internalFactorRange(T* data, int64_t startSpanIndex, int64_t endSpanIndex,
166 bool verbose) const {
167 BASPACHO_CHECK_GE(startSpanIndex, 0);
168 BASPACHO_CHECK_LE(startSpanIndex, endSpanIndex);
169 BASPACHO_CHECK_LT(endSpanIndex, (int64_t)factorSkel.spanOffsetInLump.size());
170 BASPACHO_CHECK_EQ(factorSkel.spanOffsetInLump[startSpanIndex], 0);
171 BASPACHO_CHECK_EQ(factorSkel.spanOffsetInLump[endSpanIndex], 0);
172 BASPACHO_CHECK_LE(endSpanIndex, canFactorUpTo);
173 int64_t startLump = factorSkel.spanToLump[startSpanIndex];
174 int64_t upToLump = factorSkel.spanToLump[endSpanIndex];
175
176 NumericCtxPtr<T> numCtx = symCtx->createNumericCtx<T>(maxElimTempSize, data);
177
178 for (int64_t l = 0; l + 1 < (int64_t)sparseElimRanges.size(); l++) {
179 if (sparseElimRanges[l + 1] > upToLump) {
180 BASPACHO_CHECK_EQ(sparseElimRanges[l], upToLump);
181 return;
182 } else if (startLump > sparseElimRanges[l]) {
183 BASPACHO_CHECK_GE(startLump, sparseElimRanges[l + 1]);
184 continue;
185 }
186 if (verbose) {
187 std::cout << "Elim set: " << l << " (" << sparseElimRanges[l] << ".."
188 << sparseElimRanges[l + 1] << ")" << std::endl;
189 }
190 numCtx->doElimination(*elimCtxs[l], data, sparseElimRanges[l], sparseElimRanges[l + 1]);
191 }
192
193 int64_t denseOpsFromLump = sparseElimRanges.empty() ? 0 : sparseElimRanges.back();
194 if (verbose) {
195 std::cout << "Block-Fact from: " << denseOpsFromLump << std::endl;
196 }
197
198 for (int64_t l = std::max(startLump, denseOpsFromLump);
199 l < (int64_t)factorSkel.chainColPtr.size() - 1; l++) {
200 numCtx->prepareAssemble(l);
201
202 // iterate over columns having a non-trivial a-block
203 for (int64_t rPtr = startElimRowPtr[l - denseOpsFromLump],
204 rEnd = factorSkel.boardRowPtr[l + 1] - 1; // skip last (diag block)
205 rPtr < rEnd; rPtr++) {
206 int64_t origLump = factorSkel.boardColLump[rPtr];
207 if (origLump >= upToLump) {
208 break;
209 } else if (origLump < startLump) {
210 continue;
211 }
212 eliminateBoard(*numCtx, data, rPtr);
213 }
214
215 if (l < upToLump) {
216 factorLump(*numCtx, data, l);
217 }
218 }
219}

268template <typename T>
269void Solver::internalSolveLRange(SolveCtx<T>& slvCtx, const T* matData, int64_t startSpanIndex,
270 int64_t endSpanIndex, T* vecData, int64_t stride, int nRHS) const {
271 BASPACHO_CHECK_GE(startSpanIndex, 0);
272 BASPACHO_CHECK_LE(startSpanIndex, endSpanIndex);
273 BASPACHO_CHECK_LT(endSpanIndex, (int64_t)factorSkel.spanOffsetInLump.size());
274 BASPACHO_CHECK_EQ(factorSkel.spanOffsetInLump[startSpanIndex], 0);
275 BASPACHO_CHECK_EQ(factorSkel.spanOffsetInLump[endSpanIndex], 0);
276 int64_t startLump = factorSkel.spanToLump[startSpanIndex];
277 int64_t upToLump = factorSkel.spanToLump[endSpanIndex];
278
279 int64_t denseOpsFromLump;
280 if (SparseElimSolve) {
281 for (int64_t l = 0; l + 1 < (int64_t)sparseElimRanges.size(); l++) {
282 if (sparseElimRanges[l + 1] > upToLump) {
283 BASPACHO_CHECK_EQ(sparseElimRanges[l], upToLump);
284 return;
285 } else if (startLump > sparseElimRanges[l]) {
286 BASPACHO_CHECK_GE(startLump, sparseElimRanges[l + 1]);
287 continue;
288 }
289 slvCtx.sparseElimSolveL(*elimCtxs[l], matData, sparseElimRanges[l], sparseElimRanges[l + 1],
290 vecData, stride);
291 }
292
293 denseOpsFromLump =
294 std::max(startLump, (int64_t)(sparseElimRanges.empty() ? 0 : sparseElimRanges.back()));
295 } else {
296 denseOpsFromLump = startLump;
297 }
298
299 if (factorSkel.numSpans() == factorSkel.numLumps() && slvCtx.hasFragmentedOps() && nRHS == 1) {
300 BASPACHO_CHECK_EQ(factorSkel.lumpToSpan[denseOpsFromLump], denseOpsFromLump);
301 slvCtx.fragmentedSolveL(matData, denseOpsFromLump, upToLump, vecData);
302 } else {
303 for (int64_t l = denseOpsFromLump; l < upToLump; l++) {
304 int64_t lumpStart = factorSkel.lumpStart[l];
305 int64_t lumpSize = factorSkel.lumpStart[l + 1] - lumpStart;
306 int64_t chainColBegin = factorSkel.chainColPtr[l];
307 int64_t diagBlockOffset = factorSkel.chainData[chainColBegin];
308
309 slvCtx.solveL(matData, diagBlockOffset, lumpSize, vecData, lumpStart, stride);
310
311 int64_t boardColBegin = factorSkel.boardColPtr[l];
312 int64_t boardColEnd = factorSkel.boardColPtr[l + 1];
313 int64_t belowDiagChainColOrd = factorSkel.boardChainColOrd[boardColBegin + 1];
314 int64_t numColChains = factorSkel.boardChainColOrd[boardColEnd - 1];
315 int64_t belowDiagOffset = factorSkel.chainData[chainColBegin + belowDiagChainColOrd];
316 int64_t numRowsBelowDiag =
317 factorSkel.chainRowsTillEnd[chainColBegin + numColChains - 1] -
318 factorSkel.chainRowsTillEnd[chainColBegin + belowDiagChainColOrd - 1];
319 if (numRowsBelowDiag == 0) {
320 continue;
321 }
322
323 slvCtx.gemv(matData, belowDiagOffset, numRowsBelowDiag, lumpSize, vecData, lumpStart, stride,
324 -1.0);
325
326 int64_t chainColPtr = chainColBegin + belowDiagChainColOrd;
327 slvCtx.assembleVec(chainColPtr, numColChains - belowDiagChainColOrd, vecData, stride);
328 }
329 }
330}

332template <typename T>
333void Solver::internalSolveLtRange(SolveCtx<T>& slvCtx, const T* matData, int64_t startSpanIndex,
334 int64_t endSpanIndex, T* vecData, int64_t stride,
335 int nRHS) const {
336 BASPACHO_CHECK_GE(startSpanIndex, 0);
337 BASPACHO_CHECK_LE(startSpanIndex, endSpanIndex);
338 BASPACHO_CHECK_LT(endSpanIndex, (int64_t)factorSkel.spanOffsetInLump.size());
339 BASPACHO_CHECK_EQ(factorSkel.spanOffsetInLump[startSpanIndex], 0);
340 BASPACHO_CHECK_EQ(factorSkel.spanOffsetInLump[endSpanIndex], 0);
341 int64_t startLump = factorSkel.spanToLump[startSpanIndex];
342 int64_t upToLump = factorSkel.spanToLump[endSpanIndex];
343
344 int64_t denseOpsFromLump;
345 if (SparseElimSolve) {
346 denseOpsFromLump =
347 std::max(startLump, (int64_t)(sparseElimRanges.empty() ? 0 : sparseElimRanges.back()));
348 } else {
349 denseOpsFromLump = 0;
350 }
351
352 int64_t numSpans = factorSkel.lumpToSpan[upToLump] - factorSkel.lumpToSpan[denseOpsFromLump];
353 if (numSpans == upToLump - denseOpsFromLump && slvCtx.hasFragmentedOps() && nRHS == 1) {
354 BASPACHO_CHECK_EQ(factorSkel.lumpToSpan[denseOpsFromLump], denseOpsFromLump);
355 slvCtx.fragmentedSolveLt(matData, denseOpsFromLump, upToLump, vecData);
356 } else {
357 for (int64_t l = upToLump - 1; l >= denseOpsFromLump; l--) {
358 int64_t lumpStart = factorSkel.lumpStart[l];
359 int64_t lumpSize = factorSkel.lumpStart[l + 1] - lumpStart;
360 int64_t chainColBegin = factorSkel.chainColPtr[l];
361
362 int64_t boardColBegin = factorSkel.boardColPtr[l];
363 int64_t boardColEnd = factorSkel.boardColPtr[l + 1];
364 int64_t belowDiagChainColOrd = factorSkel.boardChainColOrd[boardColBegin + 1];
365 int64_t numColChains = factorSkel.boardChainColOrd[boardColEnd - 1];
366 int64_t belowDiagOffset = factorSkel.chainData[chainColBegin + belowDiagChainColOrd];
367 int64_t numRowsBelowDiag =
368 factorSkel.chainRowsTillEnd[chainColBegin + numColChains - 1] -
369 factorSkel.chainRowsTillEnd[chainColBegin + belowDiagChainColOrd - 1];
370
371 if (numRowsBelowDiag > 0) {
372 int64_t chainColPtr = chainColBegin + belowDiagChainColOrd;
373 slvCtx.assembleVecT(vecData, stride, chainColPtr, numColChains - belowDiagChainColOrd);
374
375 slvCtx.gemvT(matData, belowDiagOffset, numRowsBelowDiag, lumpSize, vecData, lumpStart,
376 stride, -1.0);
377 }
378
379 int64_t diagBlockOffset = factorSkel.chainData[chainColBegin];
380 slvCtx.solveLt(matData, diagBlockOffset, lumpSize, vecData, lumpStart, stride);
381 }
382 }
383
384 if (SparseElimSolve) {
385 for (int64_t l = (int64_t)sparseElimRanges.size() - 2; l >= 0; l--) {
386 if (sparseElimRanges[l + 1] > upToLump) {
387 BASPACHO_CHECK_LE(sparseElimRanges[l], upToLump);
388 continue;
389 } else if (sparseElimRanges[l] < startLump) {
390 BASPACHO_CHECK_GE(startLump, sparseElimRanges[l + 1]);
391 return;
392 }
393 slvCtx.sparseElimSolveLt(*elimCtxs[l], matData, sparseElimRanges[l], sparseElimRanges[l + 1],
394 vecData, stride);
395 }
396 }
397}

399template <typename T>
400void Solver::addMvFrom(const T* matData, int64_t spanIndex, const T* inVecData, int64_t inStride,
401 T* outVecData, int64_t outStride, int nRHS, BaseType<T> alpha) const {
402 SolveCtxPtr<T> slvCtx = symCtx->createSolveCtx<T>(nRHS, matData);
403
404 BASPACHO_CHECK_GE(spanIndex, 0);
405 BASPACHO_CHECK_LT(spanIndex, (int64_t)factorSkel.spanOffsetInLump.size());
406 BASPACHO_CHECK_EQ(factorSkel.spanOffsetInLump[spanIndex], 0);
407 int64_t startFromLump = factorSkel.spanToLump[spanIndex];
408 int64_t denseOpsFromLump = startFromLump; // sparse ops not supported yet
409
410 int64_t upToLump = factorSkel.lumpStart.size() - 1;
411
412 int64_t numSpans = factorSkel.lumpToSpan[upToLump] - factorSkel.lumpToSpan[denseOpsFromLump];
413 if (numSpans == upToLump - denseOpsFromLump && slvCtx->hasFragmentedOps() && nRHS == 1) {
414 BASPACHO_CHECK_EQ(factorSkel.lumpToSpan[denseOpsFromLump], denseOpsFromLump);
415 slvCtx->fragmentedMV(matData, inVecData, denseOpsFromLump, upToLump, outVecData, alpha);
416 return;
417 }
418
419 for (int64_t l = denseOpsFromLump; l < upToLump; l++) {
420 int64_t lumpStart = factorSkel.lumpStart[l];
421 int64_t lumpSize = factorSkel.lumpStart[l + 1] - lumpStart;
422 int64_t chainColBegin = factorSkel.chainColPtr[l];
423
424 int64_t diagBlockOffset = factorSkel.chainData[chainColBegin];
425 slvCtx->symm(matData, diagBlockOffset, lumpSize, inVecData, lumpStart, inStride, outVecData,
426 outStride, alpha);
427
428 int64_t boardColBegin = factorSkel.boardColPtr[l];
429 int64_t boardColEnd = factorSkel.boardColPtr[l + 1];
430 int64_t belowDiagChainColOrd = factorSkel.boardChainColOrd[boardColBegin + 1];
431 int64_t numColChains = factorSkel.boardChainColOrd[boardColEnd - 1];
432 int64_t belowDiagOffset = factorSkel.chainData[chainColBegin + belowDiagChainColOrd];
433 int64_t numRowsBelowDiag =
434 factorSkel.chainRowsTillEnd[chainColBegin + numColChains - 1] -
435 factorSkel.chainRowsTillEnd[chainColBegin + belowDiagChainColOrd - 1];
436 if (numRowsBelowDiag == 0) {
437 continue;
438 }
439
440 int64_t chainColPtr = chainColBegin + belowDiagChainColOrd;
441 slvCtx->gemv(matData, belowDiagOffset, numRowsBelowDiag, lumpSize, inVecData, lumpStart,
442 inStride, alpha);
443 slvCtx->assembleVec(chainColPtr, numColChains - belowDiagChainColOrd, outVecData, outStride);
444
445 slvCtx->assembleVecT(inVecData, inStride, chainColPtr, numColChains - belowDiagChainColOrd);
446 slvCtx->gemvT(matData, belowDiagOffset, numRowsBelowDiag, lumpSize, outVecData, lumpStart,
447 outStride, alpha);
448 }

611SolverPtr createSolver(const Settings& settings, const std::vector<int64_t>& paramSize,
612 const SparseStructure& ss_, const std::vector<int64_t>& sparseElimRanges,
613 const unordered_set<int64_t>& elimLastIds) {
614 // no point in providing "elim last" ids if not allowing solve up to such set
615 BASPACHO_CHECK(settings.addFillPolicy == AddFillComplete || elimLastIds.empty());
616
617 BASPACHO_CHECK((int64_t)sparseElimRanges.size() != 1);
618 int64_t givenSparseElimEnd = sparseElimRanges.empty() ? 0 : sparseElimRanges.back();
619 if (!sparseElimRanges.empty()) {
620 BASPACHO_CHECK(isStrictlyIncreasing(sparseElimRanges, 0, sparseElimRanges.size()));
621 for (int64_t id : elimLastIds) {
622 BASPACHO_CHECK_GE(id, givenSparseElimEnd);
623 }
624 }
625
626 SparseStructure ss = ss_;
627 if (settings.addFillPolicy != AddFillNone) {
628 for (int64_t e = 0; e < (int64_t)sparseElimRanges.size() - 1; e++) {
629 ss = ss.addIndependentEliminationFill(sparseElimRanges[e], sparseElimRanges[e + 1]);
630 }
631 }
632
633 // create a factor where either no fill is added, either limited for given elims
634 if (settings.addFillPolicy == AddFillNone || settings.addFillPolicy == AddFillForGivenElims) {
635 vector<int64_t> spanStart;
636 spanStart.reserve(paramSize.size() + 1);
637 spanStart.insert(spanStart.end(), paramSize.begin(), paramSize.end());
638 spanStart.push_back(0);
639 cumSumVec(spanStart);
640
641 std::vector<int64_t> lumpToSpan(paramSize.size() + 1);
642 ::std::iota(lumpToSpan.begin(), lumpToSpan.end(), 0);
643
644 std::vector<int64_t> permutation(paramSize.size());
645 ::std::iota(permutation.begin(), permutation.end(), 0);
646
647 SparseStructure ssT = ss.transpose(); // to csc
648 CoalescedBlockMatrixSkel factorSkel(spanStart, lumpToSpan, ssT.ptrs, ssT.inds);
649
650 std::vector<int64_t> sparseElimRangesCopy = sparseElimRanges;
651 return SolverPtr(new Solver(std::move(factorSkel), std::move(sparseElimRangesCopy),
652 std::move(permutation), getBackend(settings),
653 settings.addFillPolicy == AddFillNone ? 0 : givenSparseElimEnd));
654 }
655
656 SparseStructure ssBottom = ss.extractRightBottom(givenSparseElimEnd);
657
658 // find best permutation for right-bottom corner that is left
659 vector<int64_t> permutation = ssBottom.fillReducingPermutation();
660 vector<int64_t> noCrossPoints;
661 if (!elimLastIds.empty()) { // force those params to go last
662 vector<int64_t> parts[2];
663 for (int64_t p : permutation) {
664 parts[elimLastIds.count(p + givenSparseElimEnd)].push_back(p);
665 }
666 noCrossPoints.push_back(parts[0].size());
667 permutation = parts[0];
668 permutation.insert(permutation.end(), parts[1].begin(), parts[1].end());
669 }
670 vector<int64_t> invPerm = inversePermutation(permutation);
671 SparseStructure sortedSsBottom = ssBottom.symmetricPermutation(invPerm, false);
672
673 // apply permutation to param size of right-bottom corner
674 std::vector<int64_t> sortedBottomParamSize(paramSize.size() - givenSparseElimEnd);
675 for (size_t i = givenSparseElimEnd; i < paramSize.size(); i++) {
676 sortedBottomParamSize[invPerm[i - givenSparseElimEnd]] = paramSize[i];
677 }
678
679 // auto select computation model (for node merge heuristic), if not provided
680 const ComputationModel* compModel = settings.computationModel ? settings.computationModel
681 : settings.backend == BackendCuda
682 ? &ComputationModel::model_Cuda117_2080Ti
683 : &ComputationModel::model_OpenBlas_i7_1185g7;
684
685 // compute as ordinary elimination tree on br-corner
686 EliminationTree et(sortedBottomParamSize, sortedSsBottom, compModel);
687 et.buildTree();
688 et.processTree(settings.findSparseEliminationRanges, noCrossPoints,
689 settings.addFillPolicy == AddFillForAutoElims);
690 et.computeAggregateStruct(settings.addFillPolicy == AddFillForAutoElims);
691
692 // ss last rows are to be permuted according to etTotalInvPerm
693 vector<int64_t> etTotalInvPerm = composePermutations(et.permInverse, invPerm);
694 vector<int64_t> fullInvPerm(givenSparseElimEnd + etTotalInvPerm.size());
695 ::std::iota(fullInvPerm.begin(), fullInvPerm.begin() + givenSparseElimEnd, 0);
696 for (size_t i = 0; i < etTotalInvPerm.size(); i++) {
697 fullInvPerm[i + givenSparseElimEnd] = givenSparseElimEnd + etTotalInvPerm[i];
698 }
699
700 // compute span start as cumSum of sorted paramSize
701 vector<int64_t> fullSpanStart(paramSize.size() + 1);
702 leftPermute(fullSpanStart.begin(), fullInvPerm, paramSize);
703 fullSpanStart[paramSize.size()] = 0;
704 cumSumVec(fullSpanStart);
705
706 // compute lump to span, knowing up to givenSparseElimEnd it's the identity
707 vector<int64_t> fullLumpToSpan;
708 fullLumpToSpan.reserve(givenSparseElimEnd + et.lumpToSpan.size());
709 fullLumpToSpan.resize(givenSparseElimEnd);
710 ::std::iota(fullLumpToSpan.begin(), fullLumpToSpan.begin() + givenSparseElimEnd, 0);
711 shiftConcat(fullLumpToSpan, givenSparseElimEnd, et.lumpToSpan.begin(), et.lumpToSpan.end());
712 BASPACHO_CHECK_EQ((int64_t)fullSpanStart.size() - 1, fullLumpToSpan.back());
713
714 // matrix with blocks not joined, we will need the first columns
715 SparseStructure sortedSsT = ss.symmetricPermutation(fullInvPerm, false).transpose();
716
717 // fullColStart joining sortedSsT.ptrs + shifted elimEndDataPtr
718 vector<int64_t> fullColStart;
719 fullColStart.reserve(givenSparseElimEnd + et.colStart.size());
720 fullColStart.insert(fullColStart.begin(), sortedSsT.ptrs.begin(),
721 sortedSsT.ptrs.begin() + givenSparseElimEnd);
722 int64_t elimEndDataPtr = sortedSsT.ptrs[givenSparseElimEnd];
723 shiftConcat(fullColStart, elimEndDataPtr, et.colStart.begin(), et.colStart.end());
724 BASPACHO_CHECK_EQ(fullColStart.size(), fullLumpToSpan.size());
725
726 // fullRowParam joining sortedSsT.inds and et.rowParam (moved)
727 vector<int64_t> fullRowParam;
728 fullRowParam.reserve(elimEndDataPtr + et.rowParam.size());
729 fullRowParam.insert(fullRowParam.begin(), sortedSsT.inds.begin(),
730 sortedSsT.inds.begin() + elimEndDataPtr);
731 shiftConcat(fullRowParam, givenSparseElimEnd, et.rowParam.begin(), et.rowParam.end());
732 BASPACHO_CHECK_EQ((int64_t)fullRowParam.size(), fullColStart.back());
733
734 CoalescedBlockMatrixSkel factorSkel(fullSpanStart, fullLumpToSpan, fullColStart, fullRowParam);
735
736 // include (additional) progressive Schur elimination sets, shifted
737 std::vector<int64_t> fullSparseElimRanges = sparseElimRanges;
738 if (!et.sparseElimRanges.empty()) {
739 shiftConcat(fullSparseElimRanges, givenSparseElimEnd,
740 et.sparseElimRanges.begin() + (sparseElimRanges.empty() ? 0 : 1),
741 et.sparseElimRanges.end());
742 }
743 if (fullSparseElimRanges.size() == 1) {
744 fullSparseElimRanges.pop_back();
745 }
746 int64_t fullSparseElimEnd = fullSparseElimRanges.empty() ? 0 : fullSparseElimRanges.back();
747
748 return SolverPtr(new Solver(
749 std::move(factorSkel), std::move(fullSparseElimRanges), std::move(fullInvPerm),
750 getBackend(settings),
751 settings.addFillPolicy == AddFillForAutoElims ? fullSparseElimEnd : paramSize.size()));
752}
